    const defEle = d.frozen ? 'Water' : d.stone ? 'Earth' : tg.element;                    // status_calc_element / _lv: Freeze = Water 1, Stone = Earth 1
    const defElv = d.frozen || d.stone ? 1 : tg.elv;
    return Object.assign({}, tg, { def1, def2, mdef2: tg.int + T(tg.vit / 2) + intD + T(vitD / 2), mdefE: mdef, flee: MX(1, tg.lv + agi), defEle, defElv,
      sizeIdx: SIZES[tg.size] == null ? 1 : SIZES[tg.size], cls: tg.boss ? 'Boss' : 'Normal', undead: defEle === 'Undead',   // undead_detect_type 0: element only
      alwaysHit: !!(d.stone || d.frozen), lex: !!d.lex });                                 // is_attack_hitting: opt1 (stone / freeze) always hit
  };

  // ---------- context ----------
  const mkCtx = (st, tgt, skill, D, set) => {
    const id = skill && skill.id, cd = id ? D.CALCDATA.skills[id] : null, sk = id ? SK[id] : null;
    const slv = id ? MX(1, MN(skill.lv | 0 || 1, (cd && cd.maxLv) || 99)) : 0;
    const c = { st, tg: effTarget(tgt), raw: tgt, skill, D, set, id, slv, cd, sk, basic: !id, flags: (cd && cd.flags) || {}, buffs: st.buffs || {}, notes: [],
      lv: x => get(st.skills, x), sp: set.sp != null ? +set.sp : num(st.maxsp), spheres: MX(0, T(+(st.buffs && st.buffs.MO_CALLSPIRITS) || 0)) };
    c.tg.race2 = Array.isArray(tgt.race2) ? tgt.race2 : race2Of(tgt.mobKey, D);                // status_get_race2(target): groups of the monster
    // hand-scoped bonus maps (Status.right.bonus / left.bonus); a map missing there falls back to the merged Status.bonus
    const scoped = (b, fb) => { const o = {}; for (const k of SCOPED) o[k] = b && b[k] !== undefined ? b[k] : fb[k]; o.noSizeFix = !!(b && b.noSizeFix); return o; };
    const rb = scoped(st.right && st.right.bonus, st.bonus || {}), lb = st.left && st.left.bonus ? scoped(st.left.bonus, {}) : {};
    // battle.conf left_cardfix_to_right (player.conf says yes, source default 0): Soulbound default = separate buckets
    c.toRight = !!set.leftCardfixToRight;
    c.hb = { right: c.toRight && st.left ? mergeBonus(rb, lb) : rb, left: c.toRight ? {} : lb };
    return c;
  };
  const sbonus = c => c.st.bonus || {};
  const weaponOk = (st, sk) => !sk.req || sk.req.indexOf(st.right ? st.right.type : 'Fist') >= 0;
  const cdAt = (c, key, dflt) => (c.cd && c.cd[key] && c.cd[key][c.slv - 1] != null ? c.cd[key][c.slv - 1] : dflt);

  // ---------- hit / crit ----------
  // is_attack_hitting (pre-re): hitrate = 80 + HIT - FLEE, skill modifyHitRate, Weaponry Research, clamp min_hitrate 5 .. max_hitrate 100
  const hitChance = (c, bonusPct) => {
    if (c.flags.IgnoreFlee || c.tg.alwaysHit) return 1;
    let h = 80 + num(c.st.hit) - c.tg.flee;
    if (c.sk && c.sk.hit) h = c.sk.hit(h, c.slv, c);
    if (bonusPct) h += bonusPct;                                    // Double Attack: + TF_DOUBLE level
    const wr = c.lv('BS_WEAPONRESEARCH');
    if (wr > 0) h += T(h * (2 * wr) / 100);                          // Weaponry Research hidden bonus
    return MN(100, MX(5, h)) / 100;
  };
  // is_attack_critical (pre-re): rnd()%1000 < cri; cri = status.cri*10 + critAddRace*10 - target LUK*2 (Status.cri includes the katar x2)
  const critChance = c => {
    if (c.sk && !c.sk.crit) return 0;
    if (c.sk && c.sk.alwaysCrit) return 1;                           // auto_counter_type default BL_ALL
    const cr = num(c.st.cri);
    if (!cr) return 0;
    const ca = sbonus(c).critAddRace;
    const cri = cr * 10 + (get(ca, c.tg.race) + get(ca, 'All')) * 10 - c.tg.luk * 2;
    return MN(1, MX(0, cri / 1000));
  };

  // ---------- weapon attack ----------
  const handInfo = (c, hn) => (hn === 'left' ? c.st.left : c.st.right);
  // battle_get_weapon_element (skill element Weapon / Endowed = weapon element, left hand uses its own weapon)
  const elementOf = (c, hn) => {
    let e = c.basic ? 'Weapon' : cdAt(c, 'element', 'Neutral');
    if (e === 'Weapon' || e === 'Endowed' || (c.sk && c.sk.weaponEle)) {                    // Soulbound: Grimtooth uses the weapon element
      const h = handInfo(c, hn); e = (h && h.element) || 'Neutral';
      const am = c.st.ammo;                                                               // arrow element (Status resolves it for bows)
      if (c.sk && c.sk.arrow && c.st.right && c.st.right.type !== 'Bow' && am && am.element && am.element !== 'Neutral' && hn === 'right') e = am.element;
    }
    if (e === 'Random') { c.notes.push('Random element skill: shown as Neutral.'); e = 'Neutral'; }
    return e;
  };
  // battle_addmastery (pre-re): masteries are added after DEF
  const mastery = (c, hn) => {
    const st = c.st, tg = c.tg, h = handInfo(c, hn), L = c.lv;
    let d = 0, s;
    if ((s = L('AL_DEMONBANE')) > 0 && (tg.undead || tg.race === 'Demon')) d += T(s * (st.baseLv / 20 + 3));
    if ((s = L('HT_BEASTBANE')) > 0 && (tg.race === 'Insect' || tg.race === 'Brute')) d += s * 4;
    switch (h ? h.type : 'Fist') {
      case '1hSword': case 'Dagger': if ((s = L('SM_SWORD')) > 0) d += s * 4; break;
      case '2hSword': if ((s = L('SM_TWOHAND')) > 0) d += s * 4; break;
      case '1hSpear': case '2hSpear': if ((s = L('KN_SPEARMASTERY')) > 0) d += s * (c.st.riding ? 5 : 4); break;      // battle.cpp:2350: x5 when mounted (Status.riding: settings.riding and KN_RIDING learned)
      case '1hAxe': case '2hAxe': if ((s = L('AM_AXEMASTERY')) > 0) d += s * 3; break;
      case 'Mace': if ((s = L('PR_MACEMASTERY')) > 0) d += s * 3; break;
      case 'Fist': case 'Knuckle': if ((s = L('MO_IRONHAND')) > 0) d += s * 3; break;
      case 'Musical': if ((s = L('BA_MUSICALLESSON')) > 0) d += s * 3; break;
      case 'Whip': if ((s = L('DC_DANCINGLESSON')) > 0) d += s * 3; break;
      case 'Book': if ((s = L('SA_ADVANCEDBOOK')) > 0) d += s * 3; break;
      case 'Katar': if ((s = L('AS_KATAR')) > 0) d += s * 3; break;
    }
    return d;
  };
  // status.cpp: forged weapon star crumbs (5 each, 3 crumbs = +40)
  const starDmg = h => { const s = ((h && +h.starCrumbs) || 0) * 5; return s >= 15 ? 40 : s; };

  // battle_calc_base_damage (pre-re) for one hand; crit = BDMG_CRIT
  const baseDamage = (c, hn, crit) => {
    const st = c.st, h = handInfo(c, hn), type = h ? h.type : 'Fist';
    let atkmax = h ? +h.watk || 0 : 0, atkmin = 0;
    const hasW = !!h && type !== 'Fist';
    const arrow = type === 'Bow' && (c.basic || !!(c.sk && c.sk.arrow));
    if (!crit || arrow) {
      atkmin = num(st.dex);
      if (hasW) atkmin = T(atkmin * (80 + (h.wlv || 1) * 20) / 100);
      if (atkmin > atkmax) atkmin = atkmax;
      if (arrow) { atkmin = T(atkmin * atkmax / 100); if (atkmin > atkmax) atkmax = atkmin; }   // bows: min = min*max/100
    }
    if (c.buffs.BS_MAXIMIZE) atkmin = atkmax;                                                     // SC_MAXIMIZEPOWER
    let d = crit ? R(atkmax, atkmax) : (atkmax > atkmin ? R(atkmin, atkmax - 1) : R(atkmin, atkmin));   // (rnd()%(max-min))+min
    const ammo = (c.st.ammo && +c.st.ammo.atk) || 0;
    if (arrow && ammo) d = rAdd(d, crit ? R(ammo, ammo) : R(0, ammo - 1));                        // arrow_atk: rnd()%arrow_atk (crit: full)
    const noSize = c.buffs.BS_WEAPONPERFECT || (st.bonus && st.bonus.noSizeFix) || (h && h.bonus && h.bonus.noSizeFix);
    if (!noSize) {                                                                                // size fix on weapon ATK only
      const row = c.D.CALCDATA.sizeFix[type];
      let sz = c.tg.size;
      if (c.st.riding && (type === '1hSpear' || type === '2hSpear') && sz === 'Medium') sz = 'Large';   // status.cpp:4238-4244: mounted with a spear, Medium counts as Large
      d = rMul(d, row && row[sz] != null ? row[sz] : 100, 100);
    }
    d = rK(d, num(st.batk));                                                                      // + status base ATK (after size fix)
    if (h && h.overMax > 0) d = rAdd(d, R(1, h.overMax));                                         // battle_add_weapon_damage: rnd()%overrefine + 1
    if (crit && +st.critAtkRate > 0) d = rRate(d, +st.critAtkRate);                               // crit_atk_rate (pre-re: before the skill ratio)
    return d;
  };
  // battle_calc_cardfix BF_WEAPON, attacker side (pre-re): per-category multiplication on a base of 1000
  const isLong = c => {
    if (c.sk && c.sk.trap) return false;
    if (c.basic) return !!c.st.right && c.st.right.type === 'Bow';
    const r = cdAt(c, 'range', 1), t = c.st.right ? c.st.right.type : 'Fist';
    return (r < 0 ? (t === 'Bow' ? 9 : /Spear/.test(t) ? 3 : 1) - (r + 1) : r) >= 4;           // battle_range_type: skill range < 4 = short
  };
  // arrow_atk (battle.cpp:7163): a bow attack (basic or an ammo skill)
  const isArrowAtk = c => !!c.st.right && c.st.right.type === 'Bow' && (c.basic || !!(c.sk && c.sk.arrow));
  // info (optional array) receives the race group part: { groups: [[name, %]], factor } for the breakdown
  const cardfixRatio = (c, hn, ignoreEle, info) => {
    const b = c.hb[hn] || {}, tg = c.tg;
    let f = 1000;
    f = T(f * (100 + get(b.addRace, tg.race) + get(b.addRace, 'All')) / 100);
    if (!ignoreEle) f = T(f * (100 + get(b.addEle, tg.defEle) + get(b.addEle, 'All')) / 100);
    f = T(f * (100 + get(b.addSize, tg.size) + get(b.addSize, 'All')) / 100);
    // addrace2 (bAddRace2, monster race groups), battle.cpp:966-968 (bow: the groups are summed, one step) and 993-994 / 1050-1051 (melee:
    // one multiplication per group the target belongs to), between size and class
    const g2 = (tg.race2 || []).map(g => [g, get(b.addRace2, g)]).filter(x => x[1]);
    if (g2.length) {
      const f0 = f;
      if (isArrowAtk(c)) f = T(f * (100 + g2.reduce((s, x) => s + x[1], 0)) / 100);
      else for (const x of g2) f = T(f * (100 + x[1]) / 100);
      if (info) info.push({ groups: g2, factor: f / f0 });
    }
    f = T(f * (100 + get(b.addClass, tg.cls) + get(b.addClass, 'All')) / 100);
    if (tg.mobKey && b.addMob && b.addMob[tg.mobKey]) f = T(f * (100 + b.addMob[tg.mobKey]) / 100);   // add_dmg by monster class
    if (isLong(c)) f = T(f * (100 + (+c.st.longAtkRate || 0)) / 100);                              // BF_LONG: long_attack_atk_rate
    return f;
  };
  // APPLY_CARDFIX: damage -= damage * (1000 - max(0, fix)) / 1000
  const applyFix = (r, f) => { const g = x => x - T(x * (1000 - MX(0, f)) / 1000); return R(g(r.min), g(r.max), r.avg - r.avg * (1000 - MX(0, f)) / 1000); };

  // battle_calc_defense_reduction (pre-re), monster target: damage * (100 - def1) / 100 - vit_def; vit_def = VIT + rnd()%((VIT/20)^2)
  const vitDefRange = v => { const m = T(v / 20) * T(v / 20); return R(v, m > 0 ? v + m - 1 : v); };
  const defPiercing = (c, hn, crit) => {                                                         // is_attack_piercing: Investigate = 2, Ice Pick = 1
    if (crit) return 0;
    if (c.sk && c.sk.pierce) return c.sk.pierce;
    const b = c.hb[hn] || {};
    return matchFlag(b.defRatioAtkRace, c.tg.race) || matchFlag(b.defRatioAtkClass, c.tg.cls) ? 1 : 0;
  };
  const ignoresDef = (c, hn, crit) => {                                                          // attack_ignores_def
    if (crit) return true;                                                                       // pre-re: criticals ignore DEF
    if (c.flags.IgnoreDefense) return true;
    const b = c.hb[hn] || {};
    return matchFlag(b.ignoreDefRace, c.tg.race) || matchFlag(b.ignoreDefClass, c.tg.cls);
  };
  const defReduce = (c, hn, crit, d, bd, label) => {
    const v = vitDefRange(c.tg.def2), def1 = MN(c.tg.def1, 100);                                 // if (def1 > 100) def1 = 100
    if (ignoresDef(c, hn, crit)) { if (bd) bd.push({ label: label + ' (DEF ignored)', value: fmtR(d) }); return d; }
    const p = defPiercing(c, hn, crit);
    const r = p ? R(T(d.min * p * (def1 + v.min) / 100), T(d.max * p * (def1 + v.max) / 100), d.avg * p * (def1 + v.avg) / 100)   // ATK_RATE(piercing*(def1+vit_def))
      : rSub(rMul(d, 100 - def1, 100), v);
    if (bd) bd.push({ label: label + (p ? ' (DEF pierce x' + p + ')' : ' (DEF ' + def1 + '%, VIT DEF ' + v.min + '-' + v.max + ')'), value: fmtR(r) });
    return r;
  };

  // One hand, one attack kind (normal / critical). Returns the damage range before the hit-count (div) fix.
  const handDamage = (c, hn, crit, bd) => {
    const st = c.st, tg = c.tg, sk = c.sk, h = handInfo(c, hn), lo = hn === 'right' ? 1 : 0;
    const L = (l, r) => { if (bd) bd.push({ label: l, value: fmtR(r) }); };
    const noMast = !!(sk && sk.noMast), refine = h ? +h.refineAtk || 0 : 0;
    // battle_calc_skill_base_damage: base ATK
    let d = baseDamage(c, hn, crit), bsd = d;                                                    // bsd = basedamage chain (Magnum Break bonus)
    if (sk && sk.shieldBase) d = R(num(st.batk), num(st.batk));                                  // Shield Boomerang: batk + 4*refine + weight/10 (shield data not in Status)
    if (sk && sk.acid) d = rK(d, -tg.def2);                                                      // Acid Terror: base ATK - target DEF2 (VIT)
    L((crit ? 'Critical ' : '') + 'base ATK (variance, size fix, status ATK, over-upgrade)', d);
    // battle_calc_attack_skill_ratio: 100 + Overthrust + Poison React + skill file ratio
    let ratio = 100;
    const ot = lvOf(c.buffs.BS_OVERTHRUST); if (ot) ratio += 5 * ot;                             // SC_OVERTHRUST val3 = 5*lv (self cast)
    if ((c.basic || c.id === 'KN_AUTOCOUNTER') && c.buffs.AS_POISONREACT) ratio += 30 * (c.lv('AS_POISONREACT') || lvOf(c.buffs.AS_POISONREACT));
    if (sk && sk.ratio) ratio = sk.ratio(ratio, c.slv, c);
    if (sk && sk.preDef) d = defReduce(c, hn, crit, d, bd, 'DEF before ratio');                  // Shield Boomerang: DEF is applied before the ratio
    if (ratio !== 100) { d = rMul(d, ratio, 100); L('Skill ratio ' + ratio + '%', d); }
    // battle_calc_skill_constant_addition
    const ca = sk && sk.constAdd ? sk.constAdd(c.slv, c) : 0;
    if (ca) { d = rK(d, ca); L('Constant skill damage +' + ca, d); }
    const sa = c.id ? get(sbonus(c).skillAtk, c.id) : 0;                                         // pc_skillatk_bonus
    if (sa) { d = rRate(d, sa); L('Skill damage bonus +' + sa + '%', d); }
    // battle_attack_sc_bonus: ATK% (pre-re multiplicative, after the ratio, before DEF; 100 for noMast skills)
    const pct = noMast ? 100 : (+c.st.atkRate || 100);
    if (pct !== 100) { d = rMul(d, pct, 100); bsd = rMul(bsd, pct, 100); L('ATK% (status effects) ' + pct + '%', d); }
    if (!(sk && sk.preDef)) d = defReduce(c, hn, crit, d, bd, 'After DEF');
    if (c.buffs.SM_MAGNUM) { const v = vitDefRange(tg.def2); bsd = rSub(rMul(bsd, 100 - MN(tg.def1, 100), 100), v); }
    // battle_calc_attack_post_defense: refine bonus (rhw.atk2), min 1; battle_calc_attack_masteries (battle_addmastery)
    if (!noMast) { d = rK(d, refine); bsd = rK(bsd, refine); L('+ refine bonus ' + refine, d); }
    d = rLo(d, lo); bsd = rLo(bsd, 1);
    if (!noMast) { const m = mastery(c, hn); d = rK(d, m); bsd = rK(bsd, m); if (m) L('+ masteries ' + m, d); d = rLo(d, lo); }
    // battle_calc_element_damage: attr_fix, then the non-elemental additions
    const ele = elementOf(c, hn);
    if (!c.flags.IgnoreElement) { d = attrFixC(c, d, ele, tg); L('Element ' + ele + ' vs ' + tg.defEle + ' ' + tg.defElv + ' (' + attrRatio(c.D, ele, tg.defEle, tg.defElv) + '%)' + (fieldPct(c, ele) ? ' +' + fieldPct(c, ele) + '% field enchant' : ''), d); }
    if (sk && sk.neutral2 && !c.flags.IgnoreElement) { d = attrFixC(c, d, 'Neutral', tg); L('Forced-neutral second attr_fix (Cart Revolution)', d); }
    const star = noMast ? 0 : starDmg(h), sph = noMast ? 0 : c.spheres * 3;                       // Star Crumbs, Spirit Spheres (3 each): after attr_fix
    if (star) { d = rK(d, star); L('+ Star Crumbs ' + star, d); }
    if (sph) { d = rK(d, sph); L('+ Spirit Spheres ' + sph, d); }
    if (sk && sk.envenom && !noMast) { d = rK(d, 15 * c.slv); d = attrFixC(c, d, ele, tg); L('Envenom +' + 15 * c.slv + ', element applied again', d); }
    if (c.buffs.SM_MAGNUM && hn === 'right' && !c.flags.IgnoreElement && !c.flags.IgnoreAtkCard) {   // SC_WATK_ELEMENT (Magnum Break): +20% of the normal attack as Fire
      const b2 = rK(attrFixC(c, bsd, 'Fire', tg), star + sph);
      d = rAdd(d, rMul(b2, 20, 100)); L('Magnum Break fire bonus (+20% of the normal attack as Fire)', d);
    }
    d = rLo(d, lo);
    // battle_calc_weapon_attack: Weaponry Research +2/lv, Hilt Binding +4 (not Cart Revolution)
    const wr = c.lv('BS_WEAPONRESEARCH'); if (wr > 0) { d = rK(d, wr * 2); L('+ Weaponry Research ' + wr * 2, d); }
    if (c.lv('BS_HILTBINDING') > 0 && !(sk && sk.noHilt)) { d = rK(d, 4); L('+ Hilt Binding 4', d); }
    // battle_calc_cardfix (attacker): race, element, size, class multiplied
    if (!c.flags.IgnoreAtkCard) {
      const info = [], f = cardfixRatio(c, hn, !!c.flags.IgnoreElement, info);
      if (f !== 1000) { d = applyFix(d, f); L('Cards / bonuses x' + (f / 1000).toFixed(3), d); }
      if (bd && info.length) bd.push({ label: 'Monster race group cards (' + info[0].groups.map(x => x[0] + ' ' + sgn(x[1]) + '%').join(', ') + '): own step, included above', value: 'x' + info[0].factor.toFixed(3) });
    }
    return d;
  };

  // battle_apply_div_fix: DAMAGE_DIV_FIX (positive: every hit deals the full damage, negative: the damage is split)
  const divFix = (r, n) => {
    if (n < 0) { const m = -n; return { total: R(T(r.min / m) * m, T(r.max / m) * m, r.avg), per: R(T(r.min / m), T(r.max / m), r.avg / m), n: m }; }
    return { total: rScale(r, n), per: r, n };
  };

  // battle_calc_weapon_attack
  const weaponAttack = c => {
    const st = c.st, sk = c.sk, bd = [], notes = c.notes, tg = c.tg;
    const r = st.right || { type: 'Fist', wlv: 0, watk: 0, refineAtk: 0, overMax: 0, element: 'Neutral', starCrumbs: 0, bonus: {} };
    const l = st.left;
    const katar = r.type === 'Katar';
    const dual = c.basic && !!l && l.type !== 'Katar';                                            // is_attack_left_handed (skills use the right hand only)
    const rightOn = !(c.basic && r.type === 'Fist' && l && l.type !== 'Fist');                    // is_attack_right_handed
    const baseHits = cdAt(c, 'hits', 1);
    const n = c.basic ? 1 : (sk.hits ? sk.hits(c.slv, baseHits, c) : baseHits);
    const blocked = !!(sk && sk.races && sk.races.indexOf(tg.race) < 0);                          // Beast Strafing: Brute / Insect only
    if (blocked) notes.push('This skill only damages ' + sk.races.join(' / ') + ' targets: no damage to this target.');
    const forced = !!(sk && sk.alwaysCrit);
    const pCrit = c.basic || (sk && sk.crit) ? critChance(c) : 0;
    const daLv = c.basic && r.type === 'Dagger' ? c.lv('TF_DOUBLE') : 0;                         // battle_calc_multi_attack: TF_DOUBLE with a dagger
    const pDA = daLv > 0 ? MN(5 * daLv, 100) / 100 : 0;
    const hit = forced ? 1 : hitChance(c, 0), hitDA = daLv > 0 ? hitChance(c, daLv) : hit;

    const run = (isCrit, rec) => {
      const o = {}, b = rec ? bd : null;
      const right = rightOn ? handDamage(c, 'right', isCrit, b) : R(0, 0);
      let left = dual ? handDamage(c, 'left', isCrit, null) : null;
      const dv = divFix(right, n);
      let rd = dv.total, per = dv.per;
      if (katar && c.basic) {                                                                    // battle_calc_attack_left_right_hands: damage2 = damage * (1 + 2*TF_DOUBLE) / 100
        const k = 1 + 2 * c.lv('TF_DOUBLE');
        o.katar = R(T(rd.min * k / 100), T(rd.max * k / 100), rd.avg * k / 100);
      }
      if (dual && /^(Thief|Assassin|Rogue)$/.test(st.job)) {                                      // AS_RIGHT 50+10*lv %, AS_LEFT 30+10*lv %
        rd = rLo(rMul(rd, 50 + 10 * c.lv('AS_RIGHT'), 100), 1);
        left = rLo(rMul(left, 30 + 10 * c.lv('AS_LEFT'), 100), 1);
        per = rd;
      }
      const lx = x => (tg.lex ? rScale(x, 2) : x);                                               // battle_calc_damage: Lex Aeterna doubles
      o.right = lx(rd); o.per = lx(per); o.left = left ? lx(left) : null; if (o.katar) o.katar = lx(o.katar);
      o.n = dv.n;
      return o;
    };
    const norm = run(forced, true);
    const critO = !forced && pCrit > 0 ? run(true, false) : null;
    const sum = o => (o.left ? rAdd(o.right, o.left) : o.right);
    const rawTotal = sum(norm);
    if (rawTotal.max < 0) notes.push('The target absorbs this element (damage <= 0).');
    const zero = R(0, 0), keep = x => (blocked ? zero : rPos(x));
    const total = keep(rawTotal);
    const katar2 = norm.katar ? keep(norm.katar) : null;
    const critTotal = critO ? keep(sum(critO)) : null;
    const nAll = total.avg + (katar2 ? katar2.avg : 0);
    const cAll = critO ? critTotal.avg + (critO.katar ? MX(0, critO.katar.avg) : 0) : 0;
    const pc = critO ? pCrit : 0;
    const daAll = daLv > 0 ? 2 * norm.right.avg + (norm.left ? norm.left.avg : 0) : 0;            // Double Attack: right hand x2 (div_ = 2)
    const avgAtk = blocked ? 0 : pDA * hitDA * daAll + (1 - pDA) * (pc * cAll + (1 - pc) * hit * nAll);   // crits always hit; DA overrides crit
    const extra = [];
    if (katar2) extra.push(Object.assign({ label: 'Katar 2nd hit (' + (1 + 2 * c.lv('TF_DOUBLE')) + '% of the 1st)', chance: 1 }, out(katar2)));
    if (daLv > 0) {
      const e = rScale(norm.right, 2);
      if (norm.left) { e.min += norm.left.min; e.max += norm.left.max; e.avg += norm.left.avg; }
      extra.push(Object.assign({ label: 'Double Attack (x2 hits)', chance: pDA }, out(e)));
    }
    const hands = dual ? [Object.assign({ hand: 'right' }, out(norm.right)), Object.assign({ hand: 'left' }, out(norm.left))] : null;
    let avgFinal = avgAtk, cycle = null;
    if (c.basic && !blocked) {                                                                    // passive procs of the basic attack
      const pAny = pDA * hitDA + (1 - pDA) * (pc + (1 - pc) * hit);                               // P(the attack lands: crits always hit)
      const pr = procExtras(c, pAny);
      for (const e of pr.rows) extra.push(e);
      avgFinal = pr.keep * avgAtk + pr.tripleAvg + pr.add;
      cycle = pr.cycle;
    }
    return { kind: 'weapon', n: norm.n, per: keep(norm.per), total, hands, extra, crit: critO ? Object.assign({ chance: pCrit }, out(critTotal)) : null,
      hit, avgAtk: avgFinal, perCast: nAll, bd, ele: elementOf(c, 'right'), forced, cycle };
  };

  // ---------- passive procs on the basic attack ----------
  // Hindsight (SA_AUTOSPELL) spell menu / level cap: skill.cpp:10710-10727 (pre-re skill_autospell) + skills/mage/hindsight.cpp:25-44 for the menu thresholds
  const HINDSIGHT = { MG_NAPALMBEAT: [1, () => 3], MG_COLDBOLT: [2, l => MN(3, l - 1)], MG_FIREBOLT: [2, l => MN(3, l - 1)], MG_LIGHTNINGBOLT: [2, l => MN(3, l - 1)],
    MG_SOULSTRIKE: [5, l => MN(3, l - 4)], MG_FIREBALL: [8, l => MN(2, l - 7)], MG_FROSTDIVER: [10, () => 1] };
  const procExtras = (c, pAny) => {
    const st = c.st, rows = [], notes = c.notes, cd = c.D.CALCDATA.skills, ctxOf = (id, lv) => mkCtx(st, c.raw, { id, lv }, c.D, c.set);
    let keep = 1, tripleAvg = 0, add = 0, cycle = null;
    // Triple Attack (battle.cpp:7267-7284, pre-re): chance 30 - lv %, before the normal attack; when it procs skill_attack(MO_TRIPLEATTACK) replaces the hit,
    // and canact_tick = tick + skill_delayfix (1000 - 4*AGI - 2*DEX) can hold back the next attack.
    const tl = c.lv('MO_TRIPLEATTACK');
    if (tl > 0 && cd.MO_TRIPLEATTACK && SK.MO_TRIPLEATTACK) {
      const p = MX(0, 30 - tl) / 100, ct = ctxOf('MO_TRIPLEATTACK', MN(tl, cd.MO_TRIPLEATTACK.maxLv)), rt = weaponAttack(ct);
      rows.push({ label: 'Triple Attack Lv ' + tl + ' (replaces the normal attack)', min: rt.total.min, max: rt.total.max, avg: rt.total.avg, chance: p * rt.hit });
      keep = 1 - p; tripleAvg = p * rt.avgAtk;
      const ad = +st.adelay || 0;
      cycle = (1 - p) * ad + p * MX(ad, delayFix(ct, cdAt(ct, 'delay', 0)));
      notes.push('Triple Attack: ' + (30 - tl) + '% chance per attack (30 - level), it replaces the normal hit (battle.cpp:7267); the triple hit uses the Raging Trifecta Blow formula, its after-delay is 1000 - 4*AGI - 2*DEX ms.');
    }
    // Auto Blitz Beat (skill.cpp:1351-1360, in skill_additional_effect: needs a landed normal attack): falcon + bow + HT_BLITZBEAT learned,
    // rnd() % 1000 <= LUK*10/3 + 1, level = min(learned, (job level + 9) / 10)
    if (st.falcon) {
      const bl = c.lv('HT_BLITZBEAT');
      if (!st.right || st.right.type !== 'Bow') notes.push('Auto Blitz Beat needs a bow.');
      else if (bl <= 0) notes.push('Auto Blitz Beat needs Blitz Beat learned.');
      else {
        const L = MN(bl, T((+st.jobLv + 9) / 10)), pr = MN(1, (T(num(st.luk) * 10 / 3) + 2) / 1000);
        const rb = miscAttack(ctxOf('HT_BLITZBEAT', L)), chance = keep * pr * pAny * rb.hit;
        rows.push({ label: 'Auto Blitz Beat Lv ' + L, min: rb.total.min, max: rb.total.max, avg: rb.total.avg, chance });
        add += chance * rb.total.avg;
        notes.push('Auto Blitz Beat: chance (LUK*10/3 + 2)/10 = ' + fmt1(pr * 100) + '% per landed attack, level min(learned ' + bl + ', (job level + 9) / 10 = ' + T((+st.jobLv + 9) / 10) + ').');
      }
    }
    // Hindsight (battle.cpp:7489-7519): every attack (hit or miss) has SC_AUTOSPELL val4 = 5 + 2*lv % to cast the spell on the target; with val3 = cap > 1
    // the cast level is rolled: 50% cap / 2, 35% cap - 1, 15% cap. SP cost skill_get_sp * 2 / 3.
    const hs = c.buffs.SA_AUTOSPELL;
    if (hs && typeof hs === 'object' && +hs.lv > 0 && HINDSIGHT[hs.skill] && cd[hs.skill] && SK[hs.skill]) {
      const hl = MN(10, +hs.lv | 0), spec = HINDSIGHT[hs.skill], cap = hl >= spec[0] ? spec[1](hl) : 0;
      if (cap > 0) {
        const learned = c.lv(hs.skill), L = learned > 0 ? MN(learned, cap) : cap;
        const dist = L > 1 ? [[L, 0.15], [L - 1, 0.35], [T(L / 2), 0.5]] : [[1, 1]];
        const byLv = {}; for (const [l, p] of dist) byLv[l] = (byLv[l] || 0) + p;
        let mn = Infinity, mx = 0, avg = 0;
        for (const l of Object.keys(byLv)) { const t = magicAttack(ctxOf(hs.skill, +l)).total; mn = MN(mn, t.min); mx = MX(mx, t.max); avg += byLv[l] * t.avg; }
        const chance = keep * (5 + 2 * hl) / 100, nm = cd[hs.skill].name;
        rows.push({ label: 'Hindsight: ' + nm + ' Lv ' + L, min: mn, max: mx, avg, chance });
        add += chance * avg;
        notes.push('Hindsight Lv ' + hl + ': ' + (5 + 2 * hl) + '% chance per attack, hit or miss (5 + 2*level, battle.cpp:7489).' + (L > 1 ? ' Cast level ' + Object.keys(byLv).sort((a, b) => b - a).map(l => l + ' (' + Math.round(byLv[l] * 100) + '%)').join(', ') + '; the damage shown is the expectation over these (battle.cpp:7496-7501).' : ''));
        notes.push('Hindsight costs 2/3 of the spell SP per cast; the spell uses the same target, MATK and element rules as a normal cast.');
        notes.push('Soulbound: Hindsight has no after-cast delay for the caster.');
        if (!learned) notes.push(nm + ' is not learned: its Hindsight cap (Lv ' + cap + ') is used.');
      }
    }
    return { rows, keep, tripleAvg, add, cycle };
  };

  // ---------- magic ----------
  // battle_calc_cardfix BF_MAGIC (pre-re): magic_addrace, magic_addele, magic_addclass
  // pre-re (battle.cpp:851-853): magic_addrace2 of every group of the target is summed INTO the race step (no separate step, unlike weapon attacks)
  const mCardfix = (c, ignoreEle, info) => {
    const b = sbonus(c), tg = c.tg;
    let f = 1000;
    const g2 = (tg.race2 || []).map(g => [g, get(b.magicAddRace2, g)]).filter(x => x[1]);
    if (g2.length && info) info.push({ groups: g2, sum: g2.reduce((s, x) => s + x[1], 0) });

    if (!ignoreEle) f = T(f * (100 + get(b.magicAddEle, tg.defEle) + get(b.magicAddEle, 'All')) / 100);
    f = T(f * (100 + get(b.magicAddClass, tg.cls) + get(b.magicAddClass, 'All')) / 100);
    return f;
  };
  // battle_calc_magic_attack
  const magicAttack = c => {
    const st = c.st, sk = c.sk, tg = c.tg, bd = [], notes = c.notes, L = (l, r) => bd.push({ label: l, value: fmtR(r) });

    const matk = ma > mi ? R(mi, ma - 1) : R(mi, mi);                                              // matk_min + rnd()%(matk_max-matk_min)
    let d, n = cdAt(c, 'hits', 1), instaKill = null;
    if (sk.hits) n = sk.hits(c.slv, n, c);
    const blocked = sk.undeadOnly && !(tg.undead || (sk.undeadOrDemon && tg.race === 'Demon'));
    if (blocked) notes.push(c.cd.name + ' only damages Undead' + (sk.undeadOrDemon ? ' / Demon' : '') + '-element targets: no damage to this target.');
    if (sk.base) {
      const v = sk.base(c.slv, c); d = R(v, v); L('Skill base damage', d);
      if (sk.turn && !tg.boss) {                                                                   // i = 20*lv + LUK + INT + base lv + 200 - 200*hp/maxhp (full HP target)
        instaKill = MN(20 * c.slv + num(st.luk) + num(st.int) + st.baseLv, 700) / 1000;
      } else if (sk.turn) notes.push('Bosses are immune to the Turn Undead instant kill.');
    } else { d = matk; L('MATK (' + mi + ' - ' + MX(mi, ma - 1) + ')', d); }
    if (sk.ratio && !sk.grand) {
      const ratio = sk.ratio(100, c.slv, c);
      if (ratio !== 100) { d = rMul(d, ratio, 100); L('Skill ratio ' + ratio + '%', d); }
    }
    if (sk.constAdd) { const k = sk.constAdd(c.slv, c); d = rK(d, k); L('Constant skill damage +' + k, d); }
    const sa = get(sbonus(c).skillAtk, c.id); if (sa) { d = rRate(d, sa); L('Skill damage bonus +' + sa + '%', d); }
    const sb = sbonus(c);
    const imdef = !!c.flags.IgnoreDefense || matchFlag(sb.ignoreMdefRace, tg.race) || matchFlag(sb.ignoreMdefClass, tg.cls);
    if (!imdef) { d = rK(rMul(d, 100 - tg.mdefE, 100), -tg.mdef2); L('After MDEF (' + tg.mdefE + '%, MDEF2 ' + tg.mdef2 + ')', d); }   // damage*(100-mdef)/100 - mdef2
    else L('MDEF ignored', d);
    if (sk.grand) {
      // Grand Cross (pre-re): ATK part = base damage after DEF + refine bonus; (ATK + MATK) * (100 + 40*lv)%
      const w = defReduce(c, 'right', false, baseDamage(c, 'right', false), null, '');
      const w2 = rLo(rK(w, st.right ? +st.right.refineAtk || 0 : 0), 1);
      d = rLo(rAdd(rLo(d, 1), w2), 1); L('+ weapon ATK part (after DEF, refine)', d);
      d = rMul(d, 100 + 40 * c.slv, 100); L('Grand Cross ratio ' + (100 + 40 * c.slv) + '%', d);
    }
    d = rLo(d, 1);
    let ele = cdAt(c, 'element', 'Neutral'); if (ele === 'Weapon' || ele === 'Endowed') ele = (st.right && st.right.element) || 'Neutral';
    if (!c.flags.IgnoreElement) { d = attrFixC(c, d, ele, tg); L('Element ' + ele + ' vs ' + tg.defEle + ' ' + tg.defElv + ' (' + attrRatio(c.D, ele, tg.defEle, tg.defElv) + '%)' + (fieldPct(c, ele) ? ' +' + fieldPct(c, ele) + '% field enchant' : ''), d); }
    if (!c.flags.IgnoreAtkCard) {
      const info = [], f = mCardfix(c, !!c.flags.IgnoreElement, info);
      if (f !== 1000) { d = applyFix(d, f); L('Magic cards / bonuses x' + (f / 1000).toFixed(3), d); }
      if (info.length) bd.push({ label: 'Monster race group magic cards (' + info[0].groups.map(x => x[0] + ' ' + sgn(x[1]) + '%').join(', ') + '): summed into the race step, included above', value: sgn(info[0].sum) + '% in the race step' });
    }
    if (sk.grand) { d = attrFixC(c, d, ele, tg); L('Grand Cross: element applied a second time', d); }
    if (d.max < 0) notes.push('The target absorbs this element (damage <= 0).');
    if (blocked) d = R(0, 0);
    const dv = divFix(d, n);
    let total = dv.total, per = dv.per;
    if (Math.abs(n) > 1) L('Hit count x' + Math.abs(n) + (n < 0 ? ' (damage split)' : ''), total);
    if (total.max > 0 && total.min < Math.abs(n)) total = R(MX(total.min, Math.abs(n)), total.max, MX(total.avg, Math.abs(n)));   // skill_min_damage (magic)
    if (tg.lex) { total = rScale(total, 2); per = rScale(per, 2); L('Lex Aeterna x2', total); }
    total = rPos(total); per = rPos(per);
    const extra = [];
    let avgAtk = total.avg;
    if (instaKill != null && !blocked) {
      extra.push({ label: 'Turn Undead instant kill', min: tg.hp, max: tg.hp, avg: tg.hp, chance: instaKill });
      avgAtk = (1 - instaKill) * total.avg + instaKill * tg.hp;
    }
    return { kind: 'magic', n: Math.abs(n), per, total, hands: null, extra, crit: null, hit: 1, avgAtk, perCast: total.avg, bd, ele };
  };

  // ---------- misc ----------
  // battle_calc_misc_attack: fixed base damage, flee check (unless IgnoreFlee), attr_fix, hit count
  const miscAttack = c => {
    const sk = c.sk, tg = c.tg, bd = [], L = (l, r) => bd.push({ label: l, value: fmtR(r) });
    const v = sk.base(c.slv, c); let d = R(v, v); L('Skill base damage', d);
    const hit = hitChance(c, 0);
    const sa = get(sbonus(c).skillAtk, c.id); if (sa) { d = rRate(d, sa); L('Skill damage bonus +' + sa + '%', d); }

    if (!c.flags.IgnoreElement) { d = attrFixC(c, d, e, tg); L('Element ' + e + ' vs ' + tg.defEle + ' ' + tg.defElv + ' (' + attrRatio(c.D, e, tg.defEle, tg.defElv) + '%)', d); }
    const n = cdAt(c, 'hits', 1);
    const dv = divFix(rLo(d, 0), n);
    let total = dv.total, per = dv.per;
    if (Math.abs(n) > 1) L('Hit count x' + Math.abs(n), total);
    if (total.max > 0 && total.min < Math.abs(n)) total = R(MX(total.min, Math.abs(n)), total.max, MX(total.avg, Math.abs(n)));   // skill_min_damage (misc)
    if (tg.lex) { total = rScale(total, 2); per = rScale(per, 2); L('Lex Aeterna x2', total); }
    return { kind: 'misc', n: Math.abs(n), per, total, hands: null, extra: [], crit: null, hit, avgAtk: hit * total.avg, perCast: total.avg, bd, ele: e };
  };

  // ---------- timing ----------
  // skill_castfix (pre-re): cast * (150 - DEX) / 150, then castrate (100 = normal) and per-skill bCastrate; no listed skill has CastTimeFlags
  const castFix = (c, base) => {
    const scale = 150 - num(c.st.dex);
    if (scale <= 0) return 0;
    let time = base * scale / 150;
    const sc = get(sbonus(c).skillCast, c.id);                                                     // bonus2 bCastrate (optional Bonus field)
    if (sc) time += time * sc / 100;
    time = time * (1 - (100 - (c.st.castRate == null ? 100 : +c.st.castRate)) / 100);
    return MX(T(time), 0);
  };
  // skill_delayfix (pre-re): delay_dependon_dex / delay_dependon_agi default 0; Monk combos: 1000 - 4*AGI - 2*DEX when no delay is set
  const delayFix = (c, base) => {
    let time = base;
    if (/^(MO_TRIPLEATTACK|MO_CHAINCOMBO|MO_COMBOFINISH)$/.test(c.id)) { if (time === 0) time = 1000; time -= 4 * num(c.st.agi) + 2 * num(c.st.dex); }
    const dr = +c.st.delayRate || 100;                                                             // Status.delayRate read as a % multiplier (100 = normal)
    if (dr !== 100) time = time * dr / 100;
    return MX(T(time), 0);
  };

  // ---------- ground / unit skills: how many ticks of one cast can hit the target ----------
  // skill.cpp:55 SKILLUNITTIMER_INTERVAL = 100: unit timers run every 100 ms and a unit hits a target again once tick >= ts->tick, ts->tick = tick + interval
  // (skill.cpp:6766-6771), so a unit interval rounds up to a multiple of 100 ms. A unit lives Duration1 ms (skill_unit_timer_sub, skill.cpp:12214: it expires
  // when tick - group->tick >= limit), so the ticks of one unit are ceil(Duration1 / rounded interval). CALCDATA.skills[id].duration / .unitInterval are
  // Duration1 and Unit.Interval of skill_db.yml. HSEL[id](c) -> {max, def, label, note}: max = ticks a standing target can take, def = the default pick.
  const UNIT_TICK = 100;
  const unitTicks = c => {
    const dur = cdAt(c, 'duration', 0), iv = (c.cd && c.cd.unitInterval) || 0;
    return dur > 0 && iv > 0 ? MX(1, Math.ceil(dur / (Math.ceil(iv / UNIT_TICK) * UNIT_TICK))) : 1;
  };
  const tickNote = (c, what) => what + ': Duration1 ' + cdAt(c, 'duration', 0) + ' ms, unit interval ' + ((c.cd && c.cd.unitInterval) || 0) + ' ms (ticks every ' + Math.ceil(((c.cd && c.cd.unitInterval) || 0) / UNIT_TICK) * UNIT_TICK + ' ms, skill.cpp:55, 6766) = at most ' + unitTicks(c) + ' ticks per target.';
  const HSEL = {
    // The 3rd hit has a 150 % freeze chance (stormgust.cpp:38); every hit knocks back 2 cells (skill_db.yml). Default 3 = hits until it freezes.
    WZ_STORMGUST: c => ({ max: unitTicks(c), def: 3, label: 'Waves that hit', note: tickNote(c, 'Storm Gust') + ' The 3rd hit has a 150% freeze chance (stormgust.cpp:38): a frozen target is Water 1 (use the Frozen debuff) and takes no more waves if it stays frozen; the default is 3.' }),
    // skill.cpp:5794 val2 = 4 + level hits per wall cell (shared by everything standing in it); each hit knocks back 2 cells unless the target is Fire / Undead element (firewall.cpp:22-26)
    MG_FIREWALL: c => { const mx = 4 + c.slv, noKb = c.tg.defEle === 'Fire' || c.tg.undead; return { max: mx, def: noKb ? mx : 1, label: 'Hits in the wall', note: 'Fire Wall: every cell hits up to 4 + level = ' + mx + ' times (skill.cpp:5794). ' + (noKb ? 'A Fire / Undead element target is not knocked back (firewall.cpp:22-26), so it keeps taking hits: default ' + mx + '.' : 'Each hit knocks the target back 2 cells (skill_db.yml), so it normally takes about 1 hit per cell it crosses: default 1.') }; },
    // meteorstorm.cpp:17-20: Duration1 / interval meteors on random cells of the (2 * 3 + 1)^2 area; each meteor hits everything within its unit range 3 once (HitCount hits)
    WZ_METEOR: c => { const iv = (c.cd && c.cd.unitInterval) || 1000, mx = MX(1, T(cdAt(c, 'duration', 0) / iv)); return { max: mx, def: MX(1, Math.round(mx * 37 * 37 / (49 * 49))), label: 'Meteors that land', note: 'Meteor Storm: ' + mx + ' meteors (Duration1 ' + cdAt(c, 'duration', 0) + ' ms / ' + iv + ' ms, meteorstorm.cpp:17), each hits ' + cdAt(c, 'hits', 1) + ' time(s). A target in the centre cell is reached by all of them; one standing at a random spot of the 7x7 area by about 57% (37/49 per axis): the default.' }; },
    WZ_VERMILION: c => ({ max: unitTicks(c), def: 3, label: 'Waves that hit', note: tickNote(c, 'Lord of Vermilion') + ' Each wave is one full damage split over ' + Math.abs(cdAt(c, 'hits', 1)) + ' hits.' }),
    PR_MAGNUS: c => ({ max: unitTicks(c), def: 3, label: 'Ticks that hit', note: tickNote(c, 'Magnus Exorcismus') + ' Each tick hits ' + cdAt(c, 'hits', 1) + ' time(s) (HitCount = level).' }),
    // skill.cpp:5819 the group counts val1 = level + 3 damaging hits shared by every target (skill.cpp:6923 sg->val1 -= 1; the group ends at 0, skill.cpp:12414)
    PR_SANCTUARY: c => { const t = unitTicks(c), mx = MN(c.slv + 3, t); return { max: mx, def: 3, label: 'Ticks that hit', note: tickNote(c, 'Sanctuary') + ' The cast can only deal level + 3 = ' + (c.slv + 3) + ' damaging ticks in total over all targets (skill.cpp:5819).' }; },
    // skill.cpp:6886-6889: with gx_allhit 0 (the compiled default, skill.conf is not in the reference copy) every cross cell hits once and then disappears
    CR_GRANDCROSS: c => ({ max: unitTicks(c), def: 1, label: 'Ticks that hit', note: tickNote(c, 'Grand Cross') + ' With gx_allhit 0 (default) each cross cell hits once (skill.cpp:6888), hence the default 1; raise it if the server enables gx_allhit.' }),
    BA_DISSONANCE: c => ({ max: unitTicks(c), def: 3, label: 'Ticks that hit', note: tickNote(c, 'Dissonance') }),
    AM_DEMONSTRATION: c => ({ max: unitTicks(c), def: 3, label: 'Ticks that hit', note: tickNote(c, 'Bomb') })
  };

  // ---------- damage taken from a monster ----------
  // battle_calc_weapon_attack with a monster source and a player target, normal attack (skill id 0), pre-renewal. Values from the shipped
  // conf/battle/battle.conf (not the compiled-in defaults where they differ): min_hitrate 5 / max_hitrate 100, agi_penalty_* and vit_penalty_*
  // (type 1, players, count 3, num 10 / 5), attack_attr_none 14 (monster normal attacks are non-elemental), enable_baseatk 0x9 (no monster
  // base ATK), enable_critical 17 (no monster crits), enable_perfect_flee 1 (players).
  const INC = { minHit: 5, maxHit: 100, agiCount: 3, agiNum: 10, vitCount: 3, vitNum: 5 };
  // is_attack_hitting 3265-3275: attacker_count >= agi_penalty_count -> flee = flee * (100 - (count - (agi_penalty_count - 1)) * agi_penalty_num) / 100, min 1
  const penFlee = (flee, n) => (n >= INC.agiCount ? MX(1, T(flee * (100 - (n - (INC.agiCount - 1)) * INC.agiNum) / 100)) : flee);
  const mobThresholds = mob => {
    const s = (mob && mob.stats) || {}, lv = +(mob && mob.lv) || 0;
    // monster HIT = level + DEX, FLEE = level + AGI (status_calc_misc 2705-2711); hitrate = 80 + HIT - FLEE (is_attack_hitting 3262, 3278), clamped 5..100
    return lv ? { hit100: lv + (+s.AGI || 0) + (INC.maxHit - 80), flee95: lv + (+s.DEX || 0) + (80 - INC.minHit) } : { hit100: 0, flee95: 0 };
  };
  // The character's DEF side against a monster hit (battle_calc_defense_reduction 4730-4883, player target, pre-re). vit_penalty (4780-4798): target_count =
  // min(count, 100 / num + count_min - 1); Steel Body keeps its hard DEF. -> {def1 (capped at 100, 4872), vit: {min,max,avg}, vn}
  const incDef = (st, mob, n, B) => {
    const bf = st.buffs || {}, sk = st.skills || {}, race = mob.race || 'Formless';
    let def1 = num(st.def), def2 = num(st.def2);
    const vn = MN(n, T(100 / INC.vitNum) + INC.vitCount - 1);

      }
      d = rLo(d, 1);
      ({ x, mod } = mixed((ele, L) => {
        let y = d, m = 100;
        if (!flags.IgnoreElement) { m = att(ele); y = rPos(rMul(y, m, 100)); L('Element ' + ele + ' vs ' + defEle + ' 1 (' + m + '%)', fmtR(y)); }
        const f = incResist(st, mob, D, ele, 'magic', ranged, flags, L === B ? B : none);
        if (f !== 1000) y = rPos(applyFix(y, f));
        if (def.grand) { y = rPos(rMul(y, att(ele), 100)); L('Element applied a second time', fmtR(y)); }   // 6256 (unconditional)
        return { x: y, mod: m };
      }));
    } else {
      const v = def.base(lv, mc, o); let d = R(v, v); B('Skill base damage', fmtR(d));
      ({ x, mod } = mixed((ele, L) => {
        let y = d, m = 100;
        const f = incResist(st, mob, D, ele, 'misc', ranged, flags, L === B ? B : none);           // BF_MISC: resists first (6587), then attr_fix (6593)
        if (f !== 1000) y = rPos(applyFix(y, f));
        if (!flags.IgnoreElement) { m = att(ele); y = rPos(rMul(y, m, 100)); L('Element ' + ele + ' vs ' + defEle + ' 1 (' + m + '%)', fmtR(y)); }
        return { x: y, mod: m };
      }));
    }
    const dv = divFix(rLo(x, 0), nHit);
    let total = dv.total, per = dv.per;
    if (dv.n > 1) B('Hit count x' + dv.n + (nHit < 0 ? ' (damage split)' : ''), fmtR(total));
    if (def.t !== 'w' && total.max > 0 && total.min < dv.n) total = R(MX(total.min, dv.n), total.max, MX(total.avg, dv.n));   // skill_min_damage 2|4 (magic, misc), battle.cpp:8370
    total = rPos(total); per = rPos(per);
    // --- landing chance
    if (always) B(def.t === 'm' ? 'Hit chance' : def.crit ? 'Hit chance (critical hits always land)' : 'Hit chance (this skill ignores Flee)', '100%');
    else {
      const raw = 80 + mobHit - flee;
      B('Monster HIT ' + mobHit + ' vs your Flee ' + flee + (flee !== num(st.flee) ? ' (' + num(st.flee) + ' with ' + n + ' attackers)' : ''), '80 + ' + mobHit + ' - ' + flee + ' = ' + raw + (def.hit ? ', skill hit modifier -> ' + (def.hit(raw, lv, mc)) : '') + ' -> ' + Math.round(hitChance * 1000) / 10 + '%');
    }
    let need95 = null;
    if (!always) for (let F = 1; F <= 5000; F++) if (rateAt(penFlee(F, n)) <= INC.minHit / 100) { need95 = F; break; }
    const procChance = def.proc || 1;
    const name = cd.name || id;
    notes.push(name + ' Lv ' + lv + ': ' + (def.t === 'w' ? 'physical (weapon) skill' : def.t === 'm' ? 'magic skill' : 'misc skill') + ' of a monster (' + (flags.TargetType || '') + ').');
    if (def.note) notes.push(def.note);
    if (def.t === 'm') notes.push('Magic always hits and is reduced by your MDEF and MDEF2; Perfect Dodge only works against normal attacks (battle.cpp:5424).');
    else if (!always || def.crit) notes.push('Perfect Dodge does not apply to skills (battle.cpp:5424); the hit rate is 80 + monster HIT - your Flee' + (def.hit ? ' with the skill modifier' : '') + ', clamped 5 - 100%.');
    if (procChance < 1) notes.push('Only ' + procChance * 100 + '% of the casts do damage (darkbreath.cpp:9).');
    if (flags.IgnoreDefense && def.t === 'w') notes.push('This skill ignores your DEF and VIT DEF.');
    notes.push('Range: ' + (ranged ? 'long' : 'short') + (def.short ? ' (always short range for this skill)' : ' (by monster range ' + (+mob.range || 1) + '; skills use the real distance, within 3 cells = short; opts.ranged overrides)') + '.');
    if (nHit !== 1 && Math.abs(nHit) > 1) notes.push('Hits per use: ' + Math.abs(nHit) + (nHit < 0 ? ', the damage is split over them' : ', every hit deals the full damage') + '. Ground and chained skills may hit again on the next tick.');
    if (n >= INC.agiCount) notes.push(n + ' attackers: Flee -' + (n - (INC.agiCount - 1)) * INC.agiNum + '% (agi_penalty).');
    notes.push('Not modelled: status effects of the skill, near-range resistance (bNearAtkDef), bSubDefEle, Assumptio, Energy Coat, Safety Wall, Kyrie Eleison, Auto Guard, Parry, Reject Sword, monster buffs (Berserk, Power Up).');
    if (total.max <= 0) notes.push('Your resistances reduce this attack to 0.');
    return { kind, skill: { id, lv, name }, hits: dv.n, perHit: out(per), min: total.min, max: total.max, avg: total.avg, hitChance, perfectDodge: 0, dodgeChance: 1 - hitChance, avgWithDodge: total.avg * hitChance,
      element: rand ? 'Random' : e, elementMod: mod, ranged, need95, breakdown: bd, notes: Array.from(new Set(notes)), perSecond: null, procChance, attackers: n };
  };

  // incoming(status, mob, D, opts): damage a monster deals to the character. opts = {attackers, ranged, attackAttrNone, mobBaseAtk, skill: {id, lv}, hp, mobHp}.
  // Without opts.skill: the normal attack (+ perSecond); with it: one use of that monster skill (null when it is not calculated, see mobAttackSkills).
  const incoming = (st, mob, D, opts) => {
    const o = opts || {};
    if (o.skill) return incomingSkill(st, mob, D, o);
    const bd = [], notes = [], B = (l, v) => bd.push({ label: l, value: v });
    const n = MX(1, T(+o.attackers || 1));                                                         // unit_counttargeted(player)
    const defEle = st.armorEle || 'Neutral', defElv = 1;                                           // players: element level 1
    const mlv = +mob.lv || 1, ms = mob.stats || {};
    const ranged = o.ranged != null ? !!o.ranged : (+mob.range || 1) >= 4;                         // battle_range_type: skillrange_by_distance (~BL_PC): <= 3 cells short, else long
    // --- base damage (battle_calc_base_damage 2526-2538, 2570-2571, 2622-2624): atkmin = atk1, atkmax = atk2, (rnd()%(max-min)) + min, + batk
    const a = Array.isArray(mob.atk) ? mob.atk : [mob.atk, mob.atk];
    const amax = +a[1] || +a[0] || 0, amin = MN(+a[0] || 0, amax);
    let d = amax > amin ? R(amin, amax - 1) : R(amin, amin);
    let batk = 0;
    if (o.mobBaseAtk) { const s = +ms.STR || 0, q = T(s / 10); batk = MN(65535, MX(0, s + q * q + T((+ms.DEX || 0) / 5) + T((+ms.LUK || 0) / 5))); }   // status_base_atk 2484-2491, only if enable_baseatk includes monsters
    d = rK(d, batk);
    B('Monster ATK ' + amin + ' - ' + amax + (batk ? ' + base ATK ' + batk : ''), fmtR(d));
    const pd = incDef(st, mob, n, B), vn = pd.vn, def1 = pd.def1, vit = pd.vit;
    d = rLo(rSub(rMul(d, 100 - def1, 100), vit), 1);                                               // damage * (100 - def1) / 100 - vit_def; battle_min_damage 1 (4931)
    B('After DEF (' + def1 + '%, VIT DEF ' + vit.min + '-' + vit.max + ')', fmtR(d));
    // --- element: monster normal attacks are Neutral (rhw.ele is never set from the monster element) and attack_attr_none skips attr_fix (3669)
    const ele = 'Neutral', none = o.attackAttrNone !== false;
    let elMod = 100;
    if (none) B('Element ' + ele + ' (attack_attr_none: monster normal attacks are non-elemental)', '100%');
    else { elMod = attrRatio(D, ele, defEle, defElv); d = rPos(rMul(d, elMod, 100)); B('Element ' + ele + ' vs ' + defEle + ' 1 (' + elMod + '%)', fmtR(d)); }
    // --- player resistances: battle_calc_cardfix BF_WEAPON target side 1089-1153, in source order, each step on a base of 1000
    const f = incResist(st, mob, D, ele, 'weapon', ranged, {}, B);
    if (f !== 1000) d = rPos(applyFix(d, f));
    // --- Defending Aura: battle_calc_damage 1832-1839, BF_LONG weapon damage, damage -= damage * val2 / 100
    const df = st.buffInfo && st.buffInfo.defender;
    if (df && ranged) { const p = +df.longReduction || 0, g = x => x - T(x * p / 100); d = R(g(d.min), g(d.max), d.avg - d.avg * p / 100); B('Defending Aura (ranged) -' + p + '%', fmtR(d)); }
    // --- landing chance: Lucky Dodge first (5424: rnd()%1000 < flee2), then is_attack_hitting 3258-3317
    const mobHit = mlv + (+ms.DEX || 0), flee = penFlee(num(st.flee), n);
    const rawHit = 80 + mobHit - flee, hitChance = MN(INC.maxHit, MX(INC.minHit, rawHit)) / 100;
    const perfectDodge = MN(1, MX(0, num(st.flee2) / 100)), land = (1 - perfectDodge) * hitChance;
    B('Monster HIT ' + mobHit + ' vs your Flee ' + flee + (flee !== num(st.flee) ? ' (' + num(st.flee) + ' with ' + n + ' attackers)' : ''), '80 + ' + mobHit + ' - ' + flee + ' = ' + rawHit + ' -> ' + Math.round(hitChance * 1000) / 10 + '%');
    B('Perfect Dodge', Math.round(perfectDodge * 1000) / 10 + '%');
    // need95: smallest status Flee whose penalised value (n attackers) drops the hit rate to the min_hitrate floor
    const need = mobHit + 80 - INC.minHit, pf = n >= INC.agiCount ? 100 - (n - (INC.agiCount - 1)) * INC.agiNum : 100;
    let need95 = null;
    if (pf > 0) { need95 = MX(1, Math.ceil(need * 100 / pf)); while (need95 > 1 && penFlee(need95 - 1, n) >= need) need95--; while (penFlee(need95, n) < need) need95++; }
    // --- damage per second: a monster's attack interval is its adelay (mob_db Delay), never below amotion (status.cpp:6417-6430: adelay = b_status->adelay capped to >= amotion,
    // aspd_rate can only shorten amotion); the next normal attack waits adelay (unit.cpp is not in the reference copy, standard rAthena). Every attacker swings on its own.
    const interval = MX(+mob.adelay || 0, +mob.amotion || 0), aps = interval > 0 ? 1000 / interval * n : 0;
    const perSecond = interval > 0 ? { min: d.min * aps, avg: d.avg * aps, max: d.max * aps, withDodge: d.avg * land * aps } : null;
    if (perSecond) notes.push('Damage per second: ' + fmt1(aps) + ' attacks per second (1000 / attack delay ' + interval + ' ms' + (n > 1 ? ' x ' + n + ' attackers' : '') + ') x the damage of one attack. An estimate: monsters also spend time walking, casting skills and being knocked back, and a stationary target is assumed.');
    notes.push('Normal attack. Monsters cannot land critical hits here (enable_critical 17 covers players and mercenaries); pick a monster skill for its numbers.');
    notes.push('Monster normal attacks are non-elemental' + (none ? ': the armor element does not change the damage (attack_attr_none 14)' : ' (shown with the Neutral table because attackAttrNone is off)') + '; the monster element is only its own defence element.');
    notes.push('Range: ' + (ranged ? 'long (monster range ' + (+mob.range || 1) + ' >= 4: attacks from more than 3 cells, long-range resistance applies)' : 'short (range ' + (+mob.range || 1) + ', within 3 cells)') + '; a monster standing within 3 cells deals short-range damage whatever its range.');
    if (n >= INC.agiCount) notes.push(n + ' attackers: Flee -' + (n - (INC.agiCount - 1)) * INC.agiNum + '% (agi_penalty), DEF -' + (vn >= INC.vitCount ? (vn - (INC.vitCount - 1)) * INC.vitNum : 0) + '% (vit_penalty).');
    if (pf <= 0) notes.push('With ' + n + ' attackers your Flee drops to 1: no Flee is enough for 95% dodge.');
    notes.push('Not modelled: near-range resistance (bNearAtkDef), bSubDefEle, Assumptio, Energy Coat, Safety Wall, Kyrie Eleison, Auto Guard, Parry, Reject Sword.');
    if (!batk) notes.push('Monsters have no base ATK / STR bonus (enable_baseatk 0x9 = players and homunculi); set opts.mobBaseAtk to add the compiled-in default.');
    if (d.max <= 0) notes.push('Your resistances reduce this attack to 0.');
    return { kind: 'weapon', skill: null, hits: 1, perHit: out(d), min: d.min, max: d.max, avg: d.avg, hitChance, perfectDodge, dodgeChance: 1 - land, avgWithDodge: d.avg * land, element: ele, elementMod: elMod,
      ranged, need95, breakdown: bd, notes: Array.from(new Set(notes)), perSecond, attackInterval: interval, attackers: n };
  };

  // ---------- public API ----------
  const attackSkills = (st, D) => {
    const res = [];
    for (const id of DAMAGE_SKILLS) {
      const lv = get(st.skills, id), cd = D.CALCDATA.skills[id], sk = SK[id];
      if (lv <= 0 || !cd) continue;
      if (D.SKT && D.SKT.skills && !D.SKT.skills[id]) continue;
      if (!weaponOk(st, sk) || (sk.shield && !st.shield)) continue;
      res.push({ id, name: cd.name, maxLv: cd.maxLv, lv: MN(lv, cd.maxLv), note: sk.undeadOnly ? 'Only damages Undead' + (sk.undeadOrDemon ? ' / Demon' : '') + '-element targets.' : undefined });
    }
    return res;
  };

  const compute = (st, target, skill, D, settings) => {
    const set = settings || {};
    const c = mkCtx(st, target, skill, D, set);
    if (skill && !(c.sk && c.cd)) throw new Error('CalcBattle: no damage formula for ' + skill.id);
    const sk = c.sk, notes = c.notes;
    const r = !skill || sk.t === 'w' ? weaponAttack(c) : sk.t === 'm' ? magicAttack(c) : miscAttack(c);
    // ground / unit skills that tick: the user picks how many ticks land (skill.hits, default from HSEL); one tick = the attack just computed
    let hitsSelectable = null, tickSpan = 0;
    const hsf = skill && HSEL[c.id];
    if (hsf) {
      const h = hsf(c), dflt = MN(h.def, h.max), req = skill.hits == null || skill.hits === '' ? dflt : Math.trunc(+skill.hits), N = MN(h.max, MX(1, isFinite(req) ? req : dflt));
      hitsSelectable = { min: 1, max: h.max, default: dflt, selected: N, label: h.label, perInstance: r.n };
      if (N > 1) { r.total = rScale(r.total, N); r.avgAtk *= N; r.perCast *= N; }
      r.n *= N;
      tickSpan = N * Math.ceil(((c.cd && c.cd.unitInterval) || 0) / UNIT_TICK) * UNIT_TICK;      // time the chosen ticks need
      notes.push(h.note);
      notes.push('The number of ' + h.label.toLowerCase() + ' depends on where the target stands and whether it moves, freezes or dies: set it to what you expect.');
      notes.push('DPS = the whole cast (' + N + ' of at most ' + h.max + ' ticks) / (cast time + the longer of the after-cast delay and the ' + tickSpan + ' ms the chosen ticks take); the unit keeps ticking while the delay runs, a second cast on top is not counted.');
    }
    if (skill) {
      if (sk.note) notes.push(sk.note);
      if (!weaponOk(st, sk)) notes.push('Needs ' + sk.req.join(' / ') + ' equipped.');
      if (sk.shield && !st.shield) notes.push('Needs a shield equipped.');
      if (sk.shieldBase) notes.push('Shield Boomerang base ATK is status ATK only: shield refine and weight are not in Status, so the damage is underestimated.');
      if (sk.riding && !st.riding) notes.push('Needs a mount (settings.riding, with Peco Peco Riding learned).');
      if (sk.arrow && st.right && st.right.type === 'Bow' && !st.ammo) notes.push('No ammo equipped.');
      if (c.id === 'MO_FINGEROFFENSIVE' && !c.spheres) notes.push('No Spirit Spheres in the Call Spirits buff.');
    } else if (st.left && st.left.type !== 'Katar') notes.push('Dual wield: the right hand uses AS_RIGHT, the left hand AS_LEFT.');
    if (c.buffs.AS_POISONREACT && (!skill || c.id === 'KN_AUTOCOUNTER')) notes.push('Poison React: the +30% per level counter attack is shown; the Envenom counter proc is not in the numbers.');
    const elementMod = c.flags.IgnoreElement ? 100 : attrRatio(D, r.ele, c.tg.defEle, c.tg.defElv);
    let castTime = 0, afterCast = 0, attackDelay;
    if (skill) {
      castTime = castFix(c, cdAt(c, 'cast', 0));
      afterCast = delayFix(c, cdAt(c, 'delay', 0));
      attackDelay = r.kind === 'magic' ? 0 : +st.amotion || 0;                                     // weapon / misc skills cannot repeat faster than the attack motion
    } else attackDelay = +st.adelay || 0;
    const cooldown = skill ? cdAt(c, 'cooldown', 0) : 0;                                            // skill cooldown runs next to the after-cast delay
    const cycle = skill ? MX(castTime + MX(afterCast, cooldown, tickSpan), attackDelay) : (r.cycle != null ? r.cycle : attackDelay);   // basic attack: average cycle incl. the Triple Attack proc delay
    const dps = cycle > 0 ? r.avgAtk / cycle * 1000 : 0;
    const hitsToKill = r.perCast > 0 ? Math.ceil(c.tg.hp / r.perCast) : Infinity;
    const bd = [{ label: 'Hit chance', value: Math.round(r.hit * 1000) / 10 + '%' }];
    if (r.crit) bd.push({ label: 'Critical chance', value: Math.round(r.crit.chance * 1000) / 10 + '%' });
    const dbg = c.tg.debuffs || {}, tgE = c.tg;
    if (dbg.stripArmor || dbg.stripShield || dbg.stripHelm || dbg.eternalChaos) {
      const parts = [];
      if (dbg.stripArmor) parts.push('Strip Armor: VIT ' + c.raw.vit + ' -> ' + (c.raw.vit - T(c.raw.vit * 40 / 100)) + ' (-40%)');
      if (dbg.stripShield) parts.push('Strip Shield: DEF ' + c.raw.def + ' -> ' + tgE.def1 + ' (-15%, before other debuffs)');
      if (dbg.stripHelm) parts.push('Strip Helm: INT ' + c.raw.int + ' -> ' + (c.raw.int - T(c.raw.int * 40 / 100)) + ' (-40%)');
      if (dbg.eternalChaos) parts.push('Eternal Chaos: VIT DEF 0');
      bd.push({ label: 'Target debuffs', value: parts.join('; ') });
      bd.push({ label: 'Target effective DEF / VIT DEF / MDEF2', value: tgE.def1 + ' / ' + tgE.def2 + ' / ' + tgE.mdef2 });
      if (dbg.eternalChaos) notes.push('Eternal Chaos sets the target VIT DEF to 0; its hard DEF is unchanged in pre-renewal (status.cpp:7756 is renewal only).');
    }
    for (const x of r.bd) bd.push(x);
    if (c.tg.lex && r.kind === 'weapon') bd.push({ label: 'Lex Aeterna x2 (on the total)', value: fmtR(r.total) });
    if (hitsSelectable) bd.push({ label: hitsSelectable.label + ' (you choose, 1 - ' + hitsSelectable.max + ')', value: hitsSelectable.selected + ' x ' + hitsSelectable.perInstance + ' hit' + (hitsSelectable.perInstance === 1 ? '' : 's') + ' each = ' + r.n + ' hits, total ' + fmtR(r.total) });
    bd.push({ label: 'Timing: cast / after-cast / cycle (ms)', value: castTime + ' / ' + afterCast + ' / ' + cycle });
    return { kind: r.kind, hits: r.n, perHit: out(r.per), total: out(r.total), hands: r.hands, extra: r.extra, crit: r.crit,
      hitChance: r.hit, avgPerAttack: r.avgAtk, hitsToKill, castTime, afterCast, attackDelay, dps,
      element: r.ele, elementMod, breakdown: bd, notes: Array.from(new Set(notes)), hitsSelectable };
  };

  return { targetFromMob, attackSkills, compute, incoming, mobAttackSkills, mobSkillList, mobThresholds, DAMAGE_SKILLS };
});
