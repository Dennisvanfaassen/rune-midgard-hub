/* calc-script.js: item effects -> Bonus (CalcScript). UMD, no DOM, no dependencies.
 *
 *   CalcScript.itemBonus(item, ctx)  -> { bonus, source: "ep8"|"rathena"|"none", applied: [text], notCalculated: [text] }
 *   CalcScript.itemBonusFrom(item, ctx, source)   force a source ("ep8" | "rathena" | "none")
 *   CalcScript.runScript(raScript, ctx) -> { bonus, applied, notCalculated }   (item scripts and combo scripts)
 *   CalcScript.emptyBonus() / CalcScript.addBonus(target, src)
 *   CalcScript.pickSource(item)      -> "ep8" | "rathena" | "none"   (Ep 8 first, rAthena fallback)
 *
 * ctx (every field optional, sane defaults):
 *   { refine, baseLv, jobLv, job (calculator job key, Novice .. Supernovice), baseStats: {str,agi,vit,int,dex,luk},
 *     skillLv(skillId) -> n, isEquipped(itemKey) -> bool (scripts pass numeric item ids: they are mapped to keys through
 *     itemById when given, and passed through unchanged otherwise), refineOf("EQI_GARMENT") -> n,
 *     equipIdOf("EQI_HAND_R") -> numeric item id | 0, hand: "right"|"left"|null (reserved: weapon-scoped placement is
 *     done by calc-status from the per-item Bonus),
 *     mobByMid?: {mid: mob}      bAddDamageClass: mob id -> mob (key, name); unknown ids are kept as key "#id"
 *     itemById?(id) -> item      lets getiteminfo(getequipid(..), ITEMINFO_VIEW) see the weapon type of an equipped id }
 *   readparam(bStr..bLuk) reads ctx.baseStats (allocated base stats, without item bonuses).
 *
 * Bonus = the spec Bonus plus these extra fields (all additive unless noted):
 *   matk          flat MATK (bMatk)
 *   skillCast     {SKILL_ID: %}  bonus2 bCastrate,"SKILL",n
 *   subLong       % less damage from ranged attacks (bLongAtkDef; Ep 8 SubRangeAttackDamage). Negative = more.
 *   critLong      Critical (display units) with ranged attacks only (bCriticalLong, e.g. Incisive Arrow)
 *   defRate, def2Rate   % change of hard DEF / soft (VIT) DEF as a delta on 100 (bDefRate -67 -> DEF x 0.33, min 0).
 *                 Ep 8 ModifyDEF_Fraction a b is mapped to round(-100 * (1 - a / b)) on both.
 * itemBonus never adds the item's own base stats (atk, def, mdef, element, weapon type): calc-status reads those from the item.
 * Special case: with source "rathena", `bonus bMdef` is skipped when the item has its own Ep 8 `mdef` field (not fromRA),
 * because rAthena scripts encode the item MDEF that way.
 *   addRace2, subRace2, magicAddRace2   {GroupName: %}  monster race groups (RC2_*: Goblin, Kobold, Orc, Golem, ...), names = CALCDATA.race2.names.
 *                 bAddRace2 / bSubRace2 / bMagicAddRace2 accept the numeric id (3) or the RC2_ constant (RC2_Orc). Player-wide (addRace2 lives in

function tokenize(src) {
  const t = [], n = src.length, idRe = /[A-Za-z_@.$#][\w@.$#]*/y, numRe = /0[xX][0-9a-fA-F]+|\d+/y;
  let i = 0;
  while (i < n) {
    const c = src[i];
    if (/\s/.test(c)) { i++; continue; }
    if (c === '/' && src[i + 1] === '/') { while (i < n && src[i] !== '\n') i++; continue; }
    if (c === '/' && src[i + 1] === '*') { const e = src.indexOf('*/', i + 2); i = e < 0 ? n : e + 2; continue; }
    const s = i;
    if (c === '"') {
      let v = ''; i++;
      while (i < n && src[i] !== '"') { if (src[i] === '\\' && i + 1 < n) { i++; v += src[i] === 'n' ? '\n' : src[i]; } else v += src[i]; i++; }
      i++; t.push({t: 'str', v, s, e: i}); continue;
    }
    if (c >= '0' && c <= '9') { numRe.lastIndex = i; const m = numRe.exec(src); i += m[0].length; t.push({t: 'num', v: Number(m[0]), s, e: i}); continue; }
    idRe.lastIndex = i;
    const m = idRe.exec(src);
    if (m) { i += m[0].length; t.push({t: 'id', v: m[0], s, e: i}); continue; }
    const two = src.substr(i, 2);
    if (['<=', '>=', '==', '!=', '&&', '||', '<<', '>>', '+=', '-=', '*=', '/=', '%='].includes(two)) { i += 2; t.push({t: 'op', v: two, s, e: i}); continue; }
    i++; t.push({t: 'op', v: c, s, e: i});
  }
  return t;
}
const BINP = {'||': 1, '&&': 2, '|': 3, '^': 4, '&': 5, '==': 6, '!=': 6, '<': 7, '>': 7, '<=': 7, '>=': 7, '<<': 8, '>>': 8, '+': 9, '-': 9, '*': 10, '/': 10, '%': 10};
const parseCache = new Map();
function parse(src) {
  if (parseCache.has(src)) return parseCache.get(src);
  const toks = tokenize(src); let p = 0;
  const isOp = v => p < toks.length && toks[p].t === 'op' && toks[p].v === v;
  const expectOp = v => { if (!isOp(v)) throw new Err('expected ' + v); p++; };
  const rawFrom = si => oneLine(src.slice(toks[si].s, toks[Math.max(si, p - 1)].e)).replace(/\s*;$/, '');
  function primary() {
    const t = toks[p++];
    if (!t) throw new Err('unexpected end');
    if (t.t === 'num') return {n: 'num', v: t.v};
    if (t.t === 'str') return {n: 'str', v: t.v};
    if (t.t === 'op' && t.v === '(') { const e = expr(); expectOp(')'); return e; }
    if (t.t === 'id') {
      if (isOp('(')) { p++; const args = []; if (!isOp(')')) { args.push(expr()); while (isOp(',')) { p++; args.push(expr()); } } expectOp(')'); return {n: 'call', name: t.v, args}; }
      return {n: 'id', name: t.v};
    }
    throw new Err('unexpected ' + t.v);
  }
  function unary() {
    const t = toks[p];
    if (t && t.t === 'op' && (t.v === '!' || t.v === '-' || t.v === '+' || t.v === '~')) { p++; return {n: 'un', op: t.v, a: unary()}; }
    return primary();
  }
  function bin(min) {
    let l = unary();
    for (;;) {
      const t = toks[p], pr = t && t.t === 'op' && BINP[t.v];
      if (!pr || pr < min) return l;
      p++; l = {n: 'bin', op: t.v, a: l, b: bin(pr + 1)};
    }
  }
  function expr() {
    const c = bin(1);
    if (isOp('?')) { p++; const a = expr(); expectOp(':'); return {n: 'tern', c, a, b: expr()}; }
    return c;
  }
  function endStmt() { if (isOp(';')) p++; else if (p < toks.length && !isOp('}')) throw new Err('expected ;'); }
  function stmt() {
    if (p >= toks.length) return null;
    const si = p, t = toks[p];
    try {
      if (t.t === 'op') {
        if (t.v === '{') { p++; const body = []; while (p < toks.length && !isOp('}')) { const s = stmt(); if (s) body.push(s); } p++; return {k: 'block', body}; }
        if (t.v === ';') { p++; return null; }
        throw new Err('unexpected ' + t.v);
      }
      if (t.t !== 'id') throw new Err('unexpected token');
      const lc = t.v.toLowerCase();
      if (lc === 'if') {
        p++; expectOp('('); const a = p, cond = expr(), b = p - 1; expectOp(')');
        const raw = 'if (' + oneLine(src.slice(toks[a].s, toks[b].e)) + ')';
        const then = stmt(); let els = null;
        if (p < toks.length && toks[p].t === 'id' && toks[p].v.toLowerCase() === 'else') { p++; els = stmt(); }
        return {k: 'if', cond, raw, then, else: els};
      }
      if (lc === 'else') throw new Err('stray else');
      p++;
      if (p < toks.length && toks[p].t === 'op' && ['=', '+=', '-=', '*=', '/=', '%='].includes(toks[p].v)) {
        const op = toks[p++].v, e = expr(); endStmt();
        return {k: 'assign', name: t.v, op, e, raw: rawFrom(si)};
      }
      const args = [];
      if (isOp('(')) { p++; if (!isOp(')')) { args.push(expr()); while (isOp(',')) { p++; args.push(expr()); } } expectOp(')'); }
      else if (!isOp(';') && p < toks.length && !isOp('}')) { args.push(expr()); while (isOp(',')) { p++; args.push(expr()); } }
      endStmt();
      return {k: 'cmd', name: t.v, args, raw: rawFrom(si)};
    } catch (e) {
      if (!(e instanceof Err)) throw e;
      if (p <= si) p = si + 1;
      let depth = 0;
      while (p < toks.length) {
        const x = toks[p];
        if (x.t === 'op' && x.v === '{') depth++;
        else if (x.t === 'op' && x.v === '}') { if (depth === 0) break; depth--; }
        else if (x.t === 'op' && x.v === ';' && depth === 0) { p++; break; }
        p++;
      }
      return {k: 'bad', raw: rawFrom(si)};
    }
  }
  const out = [];
  while (p < toks.length) { const s = stmt(); if (s) out.push(s); if (p < toks.length && isOp('}')) p++; }
  parseCache.set(src, out);
  return out;
}

const evNum = x => { if (typeof x !== 'number' || x !== x) throw new Unk('number expected'); return x; };
function evId(name, env) {
  const lc = name.toLowerCase(), c = env.ctx;
  if (/^[.@$#]/.test(name)) return lc in env.vars ? env.vars[lc] : 0;
  switch (lc) {
    case 'baselevel': return c.baseLv;
    case 'joblevel': return c.jobLv;
    case 'class': return jobId(c.job);
    case 'basejob': return jobId(c.job);
    case 'baseclass': return baseClassId(c.job);
    case 'upper': return 0;
    case 'str': case 'agi': case 'vit': case 'int': case 'dex': case 'luk': return c.baseStats[lc];
    case 'true': return 1;
    case 'false': return 0;
  }
  if (lc in K) return K[lc];
  if (lc.startsWith('eqi_')) return name.toUpperCase();
  if (/^(eff_|ig_|rc2_|sc_)/.test(lc)) return name;
  throw new Unk(name);
}
function evCall(n, env) {
  const lc = n.name.toLowerCase(), c = env.ctx, a = n.args, v = () => a.map(x => ev(x, env));
  switch (lc) {
    case 'getrefine': return c.refine;
    case 'readparam': {
      const x = a[0] && a[0].n === 'id' ? a[0].name.toLowerCase() : null;
      if (x === 'bstr' || x === 'bagi' || x === 'bvit' || x === 'bint' || x === 'bdex' || x === 'bluk') return c.baseStats[x.slice(1)];
      if (x === 'bbaselevel') return c.baseLv;
      if (x === 'bjoblevel') return c.jobLv;
      throw new Unk('readparam');
    }
    case 'getskilllv': { const s = ev(a[0], env); return c.skillLv(skId(s)) | 0; }
    case 'isequipped': return v().every(id => !!c.isEquipped(id)) ? 1 : 0;
    case 'isequippedcnt': return v().filter(id => !!c.isEquipped(id)).length;
    case 'getequiprefinerycnt': return c.refineOf(ev(a[0], env)) | 0;
    case 'getequipid': return c.equipIdOf(ev(a[0], env)) | 0;
    case 'getiteminfo': {
      const [id, what] = v();
      if (what !== K.iteminfo_view) throw new Unk('getiteminfo');
      const it = id > 0 && c.itemById ? c.itemById(id) : null;
      if (id > 0 && !c.itemById) throw new Unk('getiteminfo');
      return it && it.type === 'Weapon' ? (WTYPE[it.sub] ?? -1) : -1;
    }
    case 'eaclass': { const j = c.job === 'Supernovice' ? 'super_novice' : c.job.toLowerCase(); return EAJ[j] ?? 0; }
    case 'min': return Math.min(...v().map(evNum));
    case 'max': return Math.max(...v().map(evNum));
    case 'abs': return Math.abs(evNum(v()[0]));
  }
  throw new Unk(n.name + '()');
}
function ev(n, env) {
  switch (n.n) {
    case 'num': case 'str': return n.v;
    case 'id': return evId(n.name, env);
    case 'call': return evCall(n, env);
    case 'tern': return ev(n.c, env) ? ev(n.a, env) : ev(n.b, env);
    case 'un': { const x = ev(n.a, env); if (n.op === '!') return x ? 0 : 1; evNum(x); return n.op === '-' ? -x : n.op === '~' ? ~x : x; }
    case 'bin': {
      if (n.op === '&&') return ev(n.a, env) && ev(n.b, env) ? 1 : 0;
      if (n.op === '||') return ev(n.a, env) || ev(n.b, env) ? 1 : 0;
      const a = ev(n.a, env), b = ev(n.b, env);
      switch (n.op) {
        case '==': return a === b ? 1 : 0;
        case '!=': return a !== b ? 1 : 0;
        case '+': if (typeof a === 'string' || typeof b === 'string') return String(a) + String(b); break;
        case '<': if (typeof a === 'string' && typeof b === 'string') return a < b ? 1 : 0; break;
        case '>': if (typeof a === 'string' && typeof b === 'string') return a > b ? 1 : 0; break;
      }
      evNum(a); evNum(b);
      switch (n.op) {
        case '+': return a + b; case '-': return a - b; case '*': return a * b;
        case '/': if (!b) throw new Unk('division by zero'); return Math.trunc(a / b);
        case '%': if (!b) throw new Unk('division by zero'); return a % b;
        case '<': return a < b ? 1 : 0; case '>': return a > b ? 1 : 0; case '<=': return a <= b ? 1 : 0; case '>=': return a >= b ? 1 : 0;
        case '|': return a | b; case '&': return a & b; case '^': return a ^ b; case '<<': return a << b; case '>>': return a >> b;
      }
    }
  }
  throw new Unk('expression');
}

const IGNORED = new Set(['end', 'specialeffect', 'specialeffect2']);
function bonusCmd(arity, n, env, col) {
  const a = n.args;
  if (!a.length || a[0].n !== 'id') throw new Unk('bonus type');
  const h = H[arity][a[0].name.toLowerCase()];
  if (!h) { col.nc.push(n.raw); return; }
  h(a.slice(1).map(x => ev(x, env)), col, env.ctx);
}
function autobonusCmd(kind, n, env, col) {
  const v = n.args.map(x => ev(x, env));
  if (typeof v[0] !== 'string') throw new Unk('autobonus');
  const sub = newCol(); runInto(v[0], env.ctx, sub);
  const rate = N(v[1]) / 10, dur = N(v[2]) / 1000, bf = v[3] | 0;
  const when = kind === 2 ? hitWhen(bf) : kind === 3 ? 'when using a skill' : bf & 8 ? 'when attacking with magic' : 'on attack';
  const parts = sub.applied.map(subApplied).concat(sub.nc.map(s => 'unrecognized: ' + s));
  col.nc.push(...sub.nc);
  addText(col, `${pc(rate)}% chance ${when} for ${pc(dur)}s: ${parts.join('; ') || 'special effect'}`);
}
function runCmd(n, env, col) {
  const lc = n.name.toLowerCase();
  try {
    if (lc === 'end') { env.ended = true; return; }
    if (lc === 'set' && n.args.length === 2 && n.args[0].n === 'id') { env.vars[n.args[0].name.toLowerCase()] = ev(n.args[1], env); return; }
    if (/^bonus[2-5]?$/.test(lc)) return bonusCmd(lc === 'bonus' ? 1 : +lc[5], n, env, col);
    if (lc === 'skill') { const v = n.args.map(x => ev(x, env)); return addSkill(col, skId(v[0]), N(v[1])); }
    if (lc === 'autobonus' || lc === 'autobonus2' || lc === 'autobonus3') return autobonusCmd(lc === 'autobonus' ? 1 : +lc[9], n, env, col);
    if (IGNORED.has(lc)) return;
    col.nc.push(n.raw);
  } catch (e) { col.nc.push(n.raw); }
}
function exec(n, env, col) {
  if (!n || env.ended) return;
  switch (n.k) {
    case 'block': for (const s of n.body) { exec(s, env, col); if (env.ended) return; } break;
    case 'if': {
      let c;
      try { c = ev(n.cond, env); } catch (e) { col.nc.push(n.raw); return; }
      exec(c ? n.then : n.else, env, col);
      break;
    }
    case 'assign': {
      try {
        const k = n.name.toLowerCase(), x = ev(n.e, env);
        env.vars[k] = n.op === '=' ? x : ev({n: 'bin', op: n.op[0], a: {n: 'num', v: env.vars[k] || 0}, b: {n: 'num', v: x}}, env);
      } catch (e) { col.nc.push(n.raw); }
      break;
    }
    case 'cmd': runCmd(n, env, col); break;
    case 'bad': col.nc.push(n.raw); break;
  }
}
function runInto(src, ctx, col) {
  let ast;
  try { ast = parse(String(src || '')); } catch (e) { col.nc.push(oneLine(String(src))); return; }
  const env = {ctx, vars: {}, ended: false};
  try { for (const s of ast) { exec(s, env, col); if (env.ended) break; } } catch (e) { col.nc.push(oneLine(String(src))); }   // e.g. stack overflow on absurd nesting
}
function runScript(src, ctx) {
  const col = newCol();
  runInto(src, mkCtx(ctx), col);
  return {bonus: col.b, applied: col.applied, notCalculated: col.nc};
}

// AddExtParam / SubExtParam User VAR_*: [Bonus field, label, divisor, percent?]
const VARMAP = {VAR_STRAMOUNT: ['str', 'STR'], VAR_AGIAMOUNT: ['agi', 'AGI'], VAR_VITAMOUNT: ['vit', 'VIT'], VAR_INTAMOUNT: ['int', 'INT'], VAR_DEXAMOUNT: ['dex', 'DEX'], VAR_LUKAMOUNT: ['luk', 'LUK'],
  VAR_MAXHPAMOUNT: ['maxhp', 'Max HP'], VAR_MAXSPAMOUNT: ['maxsp', 'Max SP'], VAR_MAXHPPERCENT: ['maxhpRate', 'Max HP', 1, 1], VAR_MAXSPPERCENT: ['maxspRate', 'Max SP', 1, 1],
  VAR_ATTPOWER: ['batk', 'ATK'], VAR_ITEMDEFPOWER: ['def', 'DEF'], VAR_MDEFPOWER: ['mdef', 'MDEF'], VAR_HITSUCCESSVALUE: ['hit', 'HIT'], VAR_AVOIDSUCCESSVALUE: ['flee', 'Flee'],
  VAR_PLUSAVOIDSUCCESSVALUE: ['flee2', 'Perfect Dodge', 10], VAR_CRITICALSUCCESSVALUE: ['cri', 'Critical', 10], VAR_PLUSASPD: ['aspd', 'ASPD'], VAR_PLUSASPDPERCENT: ['aspdRate', 'ASPD', 1, 1],
  VAR_MAGICATKPERCENT: ['matkRate', 'MATK', 1, 1], VAR_HPACCELERATION: ['hpRegenRate', 'HP recovery', 1, 1], VAR_SPACCELERATION: ['spRegenRate', 'SP recovery', 1, 1]};
const STATKEYS = {STR: 'str', AGI: 'agi', VIT: 'vit', INT: 'int', DEX: 'dex', LUK: 'luk'};
const AEGIS_NULL = new Set(['DisableSkill', 'Cure', 'ResetIgnoreDEFRace', 'ResetIgnoreDEFClass', 'ResetDisappearSPAmount', 'ResetDisappearHPAmount', 'ResetInvestigate', 'ResetPushingSkill']);
// Aegis race / property names -> data names (unknown names make the line unparsable)
const aRace = x => { const r = RACEN[x]; if (!r) throw new Unk(x); return r; };
const aEle = x => { const e = PROPN[x]; if (!e) throw new Unk(x); return e; };
function aegisLine(line, col, ctx, statSeen) {
  const raw = oneLine(line);
  if (!raw) return;
  const a = raw.split(' '), c = a.shift(), sub = c.startsWith('Sub'), pm = sub ? -1 : 1, nv = i => { const x = +a[i]; if (a[i] === undefined || x !== x) throw new Unk('number'); return x; };
  const addR = (f, key, n, what) => addMap(col, f, key, n, `${sg(n)}% ${what}`);
  switch (c) {
    case 'AddExtParam': case 'SubExtParam': {
      const m = VARMAP[a[1]];
      if (a[0] !== 'User' || !m) throw new Unk(a[1]);
      if (statSeen && statSeen[m[0]]) return;                       // already given by item.stats (the data repeats some stat lines there)
      const n = nv(2) * pm / (m[2] || 1);
      return addNum(col, m[0], n, `${sg(n)}${m[3] ? '%' : ''} ${m[1]}`);
    }
    case 'EnableSkill': return a[0] === 'SA_AUTOSPELL' ? undefined : addSkill(col, a[0], nv(1));
    case 'SetAutoSpell': return a[0] !== '0' ? addText(col, `Autocasts ${skName(a[0])} on attack`) : undefined;
    case 'RaceAddDamage': case 'RaceSubDamage': return addR('addRace', aRace(a[0]), nv(1) * pm, `damage to ${aRace(a[0])}`);
    case 'AttrAddDamage': case 'AttrAddDamageSelf': case 'AttrSubDamage': return addR('addEle', aEle(a[0]), nv(1) * pm, `damage to ${aEle(a[0])} element`);
    case 'AddDamage_Size': if (a[0] === 'Target') return addR('addSize', a[1], nv(2), `damage to ${a[1]} size`); return addR('subSize', a[1], -nv(2), `resistance to ${a[1]} size`);
    case 'ClassAddDamage': {
      const cl = a[0] === '1' ? 'Boss' : 'Normal', n = nv(2);
      return a[1] === 'Target' ? addR('addClass', cl, n, `damage to ${groupName('class', cl)}`) : addR('subClass', cl, -n, `resistance to ${groupName('class', cl)}`);
    }
    case 'NameAddDamage': {
      const n = nv(2);
      return a[1] === 'Target' ? addMap(col, 'addMob', a[0], n, `${sg(n)}% damage to ${(ctx.mobByKey && ctx.mobByKey[a[0]] && ctx.mobByKey[a[0]].name) || a[0]}`) : addText(col, `${sg(n)}% damage taken from ${a[0]}`);
    }
    case 'AddCRIPercent_Race': { const n = nv(1) / 10; return addMap(col, 'critAddRace', aRace(a[0]), n, `${sg(n)} Critical vs ${aRace(a[0])}`); }
    case 'SetIgnoreDEFRace': return addFlagMap(col, 'ignoreDefRace', aRace(a[0]), `Ignores DEF of ${aRace(a[0])}`);
    case 'SetIgnoreDEFClass': { const cl = a[0] === 'CLASS_Boss' ? 'Boss' : 'Normal'; return addFlagMap(col, 'ignoreDefClass', cl, `Ignores DEF of ${groupName('class', cl)}`); }
    case 'AddAttrTolerace': case 'SubAttrTolerace': return addR('subEle', aEle(a[0]), nv(1) * pm, `resistance to ${aEle(a[0])} element`);
    case 'AddRaceTolerace': case 'SubRaceTolerace': return addR('subRace', aRace(a[0]), nv(1) * pm, `resistance to ${aRace(a[0])}`);
    case 'RaceAddDamageSelf': return addR('subRace', aRace(a[0]), -nv(1), `resistance to ${aRace(a[0])}`);
    case 'AddRangeAttackDamage': case 'SubRangeAttackDamage': { const n = nv(1) * -pm; return addNum(col, 'subLong', n, `${sg(n)}% resistance to ranged attacks`); }
    case 'BodyAttribute': { const e = aEle(a[0]); col.b.defEle = e; return col.applied.push(`Armor element becomes ${e}`); }
    case 'AddSpellCastTime': case 'SubSpellCastTime': { const n = nv(0) * pm; return addNum(col, 'castRate', n, `${sg(n)}% cast time`); }
    case 'AddSPconsumption': case 'SubSPconsumption': { const n = nv(0) * pm; return addNum(col, 'useSpRate', n, `${sg(n)}% SP cost`); }
    case 'ModifyDEF_Fraction': {
      const x = nv(0), y = nv(1);
      if (!(x < y)) return;
      const n = Math.round(-100 * (1 - x / y));
      col.b.defRate += n; col.b.def2Rate += n; return col.applied.push(`${sg(n)}% DEF (hard and soft)`);
    }
    case 'NoDispell': if (a[0] === '1') { col.b.noCastCancel = true; col.applied.push('Casting cannot be interrupted'); } return;
    case 'NoJamstone': if (a[0] === '1') { col.b.noGemstone = true; col.applied.push('Skills need no gemstones'); } return;
    case 'PerfectDamage': if (a[0] === '1') { col.b.noSizeFix = true; col.applied.push('Ignores size penalty'); } return;
    case 'IndestructibleWeapon': return a[0] === '1' ? addText(col, 'Weapon cannot be broken') : undefined;
    case 'IndestructibleArmor': return a[0] === '1' ? addText(col, 'Armor cannot be broken') : undefined;
    case 'Magicimmune': return a[0] === '1' ? addText(col, 'Immune to magic') : undefined;
    case 'Reincarnation': return a[0] === '1' ? addText(col, 'Full HP and SP on respawn') : undefined;
    case 'SplashAttack': return a[0] === '1' ? addText(col, 'Attacks hit surrounding targets') : undefined;
    case 'Condition': {
      const dur = a[1] === '9999' ? 'permanently' : `for ${a[1]} s`, s = a[0];
      if (s === 'EFFECTHaste') return addText(col, `Movement speed +25% ${dur}`);
      if (s === 'EFFECTEndure') return addText(col, `Endure effect ${dur}`);
      if (s === 'EFFECTHasteAttack') return addText(col, `ASPD +${a[2]}% ${dur} (timed effect, not added to ASPD)`);
      if (s === 'EFFECTPlusAttackPower') return addText(col, `ATK +${a[2]} ${dur}`);
      if (s === 'EFFECTPlusMagicPower') return addText(col, `MATK +${a[2]} ${dur}`);
      if (s === 'EFFECTWedding') return addText(col, `Wedding effect ${dur}`);
      return addText(col, `${a[2]}% chance to be ${nm(STATEN, s)} ${dur}`);
    }
    case 'AddStateAttack': return addText(col, `${pc(nv(1) / 10)}% chance to inflict ${nm(STATEN, a[0])} on attack`);
    case 'AddAtkStateSelf': return addText(col, `${pc(nv(1) / 10)}% chance to inflict ${nm(STATEN, a[0])} on yourself when attacking`);
    case 'AddStateTolerace': case 'SubStateTolerace': return addText(col, `${sg(nv(1) * pm)}% resistance to ${nm(STATEN, a[0])}`);
    case 'AddAtk_Coma': return addText(col, `${pc(nv(0) / 100)}% chance to cause Coma on attack`);
    case 'AddAtk_ComaRace': return addText(col, `${pc(nv(1) / 100)}% chance to cause Coma to ${aRace(a[0])}`);
    case 'AddAtk_DamageWeapon': return addText(col, `${nv(1)}% chance to break the target's weapon`);
    case 'AddAtk_DamageArmor': return addText(col, `${nv(1)}% chance to break the target's armor`);
    case 'AddHPdrain': return addText(col, `${nv(0)}% chance to drain ${nv(1)}% of damage as HP`);
    case 'AddSPdrain': return addText(col, `${nv(0)}% chance to drain ${nv(1)}% of damage as SP`);
    case 'AddDestroySP_Attack': return addText(col, `${nv(1)}% chance to drain ${nv(2)}% of the target's SP`);
    case 'AddSPAmount_Action': case 'SubSPAmount_Action': return addText(col, `${sub ? 'Loses' : 'Gains'} ${nv(1)} SP per attack`);
    case 'AddSPAmount_Attack': return addText(col, `Gains ${nv(1)} SP per hit`);
    case 'AddSPAmount_AtkRace': return addText(col, `Gains ${nv(2)} SP per hit on ${aRace(a[0])}`);
    case 'AddSPAmount_Kill': return addText(col, `Gains ${nv(0)} SP per kill`);
    case 'AddSPAmount_KillRace': return addText(col, `Gains ${nv(1)} SP per ${aRace(a[0])} kill`);
    case 'AddEXPPercent_KillRace': return addText(col, `+${nv(1)}% EXP from ${aRace(a[0])}`);
    case 'SetDisappearSPAmount': return addText(col, `Loses ${nv(2)} SP every ${nv(1)} seconds`);
    case 'SetDisappearHPAmount': return addText(col, `Loses ${nv(2)} HP every ${nv(1)} seconds`);
    case 'AddMeleeAttackReflect': return addText(col, `Reflects ${nv(0)}% of melee damage`);
    case 'AddChangeMonster': return addText(col, `${nv(0)}% chance to transform the target into another monster`);
    case 'SetPushingSkill': return addText(col, `${skName(a[0])} Lv ${nv(1)} knocks the target back`);
    case 'AddGuideAttack': return addText(col, `+${nv(0)}% perfect hit`);
    case 'AddReceiveItem_Race': return addText(col, `${nv(2)}% chance to get ${a[1].replace(/_/g, ' ')} from ${aRace(a[0])}`);
    case 'SetInvestigate': return addFlagMap(col, 'defRatioAtkClass', 'All', 'Damage depends on the DEF of all monsters (like Investigate)');   // same as rAthena bDefRatioAtkClass,Class_All
  }
  if (AEGIS_NULL.has(c)) return;
  throw new Unk(c);
}
function ep8Into(item, ctx, col) {
  const st = item.stats || {}, seen = {};
  for (const k in STATKEYS) if (st[k]) { seen[STATKEYS[k]] = true; addNum(col, STATKEYS[k], st[k], `${sg(st[k])} ${k}`); }
  const sc = item.script;
  if (!sc) return;
  for (const ev of Object.keys(sc)) for (const line of sc[ev]) {
    try { aegisLine(line, col, ctx, seen); } catch (e) { col.nc.push(oneLine(line)); }
  }
}

/* ---------- items ---------- */
function pickSource(item) {
  if (!item) return 'none';
  const sc = item.script, hasLines = sc && Object.keys(sc).some(k => Array.isArray(sc[k]) && sc[k].length);
  if (hasLines || (item.stats && Object.keys(item.stats).length)) return 'ep8';
  if (item.ra && item.ra.script && String(item.ra.script).trim()) return 'rathena';
  return 'none';
}
function itemBonusFrom(item, ctx, source) {
  const c = mkCtx(ctx), col = newCol();
  if (item && source === 'ep8') ep8Into(item, c, col);
  else if (item && source === 'rathena' && item.ra && item.ra.script) {
    // rAthena expresses item MDEF as `bonus bMdef`; an Ep 8 item carries it in item.mdef (read by calc-status), so do not count it twice
    c.skipMdef = typeof item.mdef === 'number' && !item.fromRA;
    runInto(item.ra.script, c, col);
  }
  return {bonus: col.b, source, applied: col.applied, notCalculated: col.nc};
}
const itemBonus = (item, ctx) => itemBonusFrom(item, ctx, pickSource(item));

return {emptyBonus, addBonus, runScript, itemBonus, itemBonusFrom, pickSource, fields: {num: NUMF, map: MAPF, bool: BOOLF}};
