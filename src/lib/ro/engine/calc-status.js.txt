const MANUAL_NUM = [['str', 'STR'], ['agi', 'AGI'], ['vit', 'VIT'], ['int', 'INT'], ['dex', 'DEX'], ['luk', 'LUK'], ['allStats', 'All stats'], ['batk', 'ATK'], ['watk', 'Weapon ATK'], ['matk', 'MATK'],

const sgn = n => (n > 0 ? '+' : '') + n;
// -> {bonus, applied}: the sparse object turned into a Bonus plus readable lines (unknown keys and non-numbers are ignored)
function manualBonus(m, skillName) {
  const b = CalcScript.emptyBonus(), applied = [];
  if (!m || typeof m !== 'object') return {bonus: b, applied};
  for (const [k, label] of MANUAL_NUM) {
    const n = Number(m[k]); if (!isFinite(n) || !n) continue;
    b[k] = n; applied.push(label.charAt(0) === '%' ? `${sgn(n)}${label}` : `${sgn(n)} ${label}`);
  }
  for (const [k, what] of MANUAL_MAP) {
    const src = m[k]; if (!src || typeof src !== 'object') continue;
    for (const key in src) {
      const n = Number(src[key]); if (!isFinite(n) || !n) continue;
      b[k][key] = n;
      const tail = k === 'addEle' || k === 'subEle' ? key + ' element' : k === 'addSize' ? key + ' size' : k === 'addClass' ? key + ' class' : key;
      applied.push(k === 'skillAtk' ? `${sgn(n)}% ${skillName ? skillName(key) : key} damage` : `${sgn(n)}% ${what} ${tail}`);
    }
  }
  return {bonus: b, applied};
}

  }
  const bv = id => typeof bf[id] === 'number' ? bf[id] : 0;

   *  addRace/addEle/addSize/addClass : lr 0 and 2 -> right_weapon, 1 -> left_weapon, 3 -> arrow_add* (pc.cpp:4503-4535)

  if (bv('BS_ADRENALINE2')) hasteSkill('BS_ADRENALINE2', can('BS_ADRENALINE2') ? 300 : 200, rType === 'Bow' ? [] : null, 'a non-bow weapon');
  if (song.cross && !ranged && rType !== 'Bow') haste.push({id: 'BA_ASSASSINCROSS', val: song.cross.aspd});   // 8453-8467: not with bows
  let hasteVal = 0; for (const h of haste) hasteVal = MX(hasteVal, h.val);
  rate -= hasteVal;
  const pot = bon('POT_BERSERK') ? 200 : bon('POT_AWAKENING') ? 150 : bon('POT_CONCENTRATION') ? 100 : 0;   // SC_ASPDPOTION0..2: val2 = 50 * (2 + n) (11135), stacks with the skill
  rate -= pot;
  if (bon('MO_STEELBODY')) rate += 250;                                              // 8491
  const defender = bv('CR_DEFENDER') > 0 && !!(lItem && lItem.type === 'Armor');
  if (bv('CR_DEFENDER') && !defender) notes.push('Defending Aura needs a shield (pc.cpp:7458-7461 ends it without one): not counted.');
  const defLv = defender ? bv('CR_DEFENDER') : 0;
  if (defender) rate += 250 - 50 * defLv;                                            // 8493: SC_DEFENDER val4 = 250 - 50 * lv (11318)
  rate = clamp(rate, 0, 32767);
  let amotion = T(amo * rate / 1000);
  amotion = clamp(amotion, 100, 4000);                                               // max_aspd 190 -> 100 (battle.cpp:8965), MIN_ASPD 8000 / 2
  const adelay = amotion * 2, aspd = (2000 - amotion) / 10;

  /* --- cast / delay (skill_castfix 10122-10190, skill_castfix_sc 10205-10220, skill_delayfix 10462-10490) --- */
  const castrate = MX(0, 100 + main.castRate);                                      // sd->castrate (bCastrate adds)
  let castRate = MX(0, castrate - (song.bragi ? song.bragi.cast : 0));
  if (bv('PR_SUFFRAGIUM')) castRate = castRate * (100 - 15 * bv('PR_SUFFRAGIUM')) / 100;   // 11791 (pre-renewal val2 = 15 * lv)
  const delayRed = (song.bragi ? song.bragi.delay : 0) - main.delayRate;           // bDelayRate: bonus.delayrate -= val
  const delayRate = MX(0, 100 - MN(delayRed, 100));
  const spCostRate = MX(0, 100 + main.useSpRate - (song.service ? song.service.cost : 0));   // dsprate (status.cpp:4662)

  /* --- natural regeneration (status_calc_regen 5264-5372, status_calc_regen_rate 5377-5509, status_natural_heal 15444-15640) --- */
  const regen = (() => {
    const HP_IV = 6000, SP_IV = 8000, SK_IV = 10000;                                // conf/battle/player.conf:57, 60, 63 (battle.cpp:8387-8389 defaults)
    const hpRec = MX(0, 100 + main.hpRegenRate), spRec = MX(0, 100 + main.spRegenRate);   // sd->hprecov_rate / sprecov_rate start at 100 (3788), bHPrecovRate adds (pc.cpp:3986), min 0 (4670)
    let hp = T(s.vit / 5) + MX(1, T(maxhp / 200));                                  // 5276
    if (hpRec !== 100) hp = T(hp * hpRec / 100);
    hp = clamp(hp, 0, 32767);                                                       // players: min 0 (5281-5283)
    let sp = 1 + T(s.int / 6) + T(maxsp / 100);                                     // 5285
    if (s.int >= 120) sp += T((s.int - 120) / 2) + 4;
    if (spRec !== 100) sp = T(sp * spRec / 100);
    sp = clamp(sp, 0, 32767);
    if (sk('HP_MEDITATIO') > 0) sp = clamp(T(sp * (100 + 3 * sk('HP_MEDITATIO')) / 100), 1, 32767);   // 5295-5298
    const rn = [];
    const spRate = bv('PR_MAGNIFICAT') ? 200 : 100;                                 // SC_MAGNIFICAT: regen->rate.sp += 100 (5439); the skill regen (sregen) rate stays 100
    const hpIv = HP_IV, spIv = T(SP_IV / (spRate / 100));                           // status_natural_heal 15548, 15583: interval = base / (rate / 100 * multi), multi 1 standing
    // skill regen (sregen): SM_RECOVERY / MG_SRECOVERY (5305-5322), runs every natural_heal_skill_interval, not while walking
    const skillHp = sk('SM_RECOVERY') > 0 ? sk('SM_RECOVERY') * 5 + T(sk('SM_RECOVERY') * maxhp / 500) : 0;
    const skillSp = sk('MG_SRECOVERY') > 0 ? sk('MG_SRECOVERY') * 3 + T(sk('MG_SRECOVERY') * maxsp / 500) : 0;
    // sitting skill regen (ssregen): MO_SPIRITSRECOVERY (5328-5343), only sitting (status_natural_heal 15506: vd->dead_sit == 2), no sphere condition in the source
    const sr = sk('MO_SPIRITSRECOVERY');
    const sitHp = sr > 0 ? sr * 4 + T(sr * maxhp / 500) : 0, sitSp = sr > 0 ? sr * 2 + T(sr * maxsp / 500) : 0;
    // no natural SP regen: Maximize Power, and a Monk in Fury (5421-5428)
    const spBlocked = bon('BS_MAXIMIZE') || (job === 'Monk' && bv('MO_EXPLOSIONSPIRITS') > 0);
    if (spBlocked) rn.push(bon('BS_MAXIMIZE') ? 'Maximize Power stops natural SP regeneration (status.cpp:5421).' : 'Fury stops natural SP regeneration for a Monk (status.cpp:5421-5424).');
    rn.push('Intervals: HP 6000 ms, SP 8000 ms, skill regeneration 10000 ms (conf/battle/player.conf:57-63); the server ticks on a 500 ms grid.');
    rn.push('Standing still is assumed, not sitting. Sitting halves the HP and SP intervals (status.cpp:15548); the sitting-only skill regeneration is listed as sitHp / sitSp.');
    rn.push('No natural regeneration at 50% weight or more (battle.cpp:8394 natural_heal_weight_rate, pre-renewal).');
    rn.push('Skill regeneration (Increase HP / SP Recovery) pauses while walking (status.cpp:15525).');
    if (bv('PR_MAGNIFICAT')) rn.push('Magnificat doubles the natural SP regeneration rate (status.cpp:5439): SP interval 4000 ms; it does not affect the skill regeneration.');
    if (sk('SM_MOVINGRECOVERY') > 0) rn.push('Moving HP-Recovery: natural HP regeneration also runs while walking, at half speed (status.cpp:4653, 15529, 15547). Not included in hp / hpInterval.');
    else rn.push('Without Moving HP-Recovery there is no natural HP regeneration while walking.');
    if (sr > 0) rn.push('Spiritual Cadence (Spirits Recovery): HP / SP every 10000 ms while sitting; the source has no spirit sphere condition.');
    if (hpRec !== 100 || spRec !== 100) rn.push(`Item recovery rates: HP ${hpRec}%, SP ${spRec}% (bHPrecovRate / bSPrecovRate).`);
    return {hp, hpInterval: hpIv, sp: spBlocked ? 0 : sp, spInterval: spIv, skillHp, skillHpInterval: SK_IV, skillSp, skillSpInterval: SK_IV, sitHp, sitSp, sitInterval: SK_IV, spBlocked, notes: rn};
  })();

  /* --- Status --- */
  const out = {};
  const stat = k => ({base: bs[k], bonus: s[k] - bs[k], total: s[k]});
  const hand = (h) => {
    const x = hands[h]; if (!x) return null;
    const w = x.w, bn = h === 'right' ? CalcScript.addBonus(CalcScript.emptyBonus(), main) : CalcScript.addBonus(CalcScript.emptyBonus(), acc.left);
    return {type: h === 'right' ? rType : lType, wlv: w ? w.wlv : 0, watk: x.watk, refineAtk: x.refineAtk, overMax: x.overMax, element: eleOf(h),
      starCrumbs: w && w.forged ? w.crumbs : 0, star: w ? w.star : 0, forged: !!(w && w.forged), bonus: bn};
  };
  Object.assign(out, {job, baseLv, jobLv});
  for (const k of ST) out[k] = stat(k);
  Object.assign(out, {statPointsUsed: statPointsUsed(bs), statPointsTotal: CD.statPoints[baseLv - 1], maxhp, maxsp, batk});
  out.right = hand('right'); out.left = hand('left');
  out.shield = !!(lItem && lItem.type === 'Armor');
  out.shieldItem = out.shield ? {key: lItem.key, refine: clamp(eq.left.refine | 0, 0, 10), weight: lItem.weight | 0} : null;
  out.dualWield = dualWield; out.twoHanded = !!(rItem && rItem.loc === 'Two-handed Weapon');
  if (ammo) { ammo.hit = arrowHit; ammo.cri = arrowCri; }
  out.ammo = ammo;
  const mainOut = CalcScript.addBonus(CalcScript.emptyBonus(), main);
  Object.assign(out, {matkMin, matkMax, def: defV, def2, mdef, mdef2, hit, flee, flee2: flee2I / 10, cri: clamp(katar ? criI * 2 : criI, 10, 32767) / 10, criBase: criI / 10, amotion, adelay, aspd, castRate, delayRate, spCostRate,
    atkRate: 100 + (provLv ? 2 + 3 * provLv : 0),   // battle_get_atkpercent (battle.cpp:4355): Provoke / Auto Berserk only; bAtkRate stays in bonus
    matkRate: matkRateMul, critAtkRate: main.critAtkRate, longAtkRate: main.longAtkRate, bonus: mainOut, skills});
  // status.cpp:6147-6150: katar doubles the crit inside status, so `cri` already includes it (criBase = without)
  const echo = {};
  for (const id in bf) echo[id] = bf[id];
  if (['BA_WHISTLE', 'DC_HUMMING', 'BA_POEMBRAGI', 'BA_ASSASSINCROSS', 'DC_FORTUNEKISS', 'DC_SERVICEFORYOU', 'BA_APPLEIDUN'].some(id => bf[id])) echo.SONG_CASTER = sc;
  out.buffs = echo;
  out.buffInfo = {haste: hasteVal, hasteFrom: haste.filter(h => h.val === hasteVal).map(h => h.id), potion: pot, aspdRate: rate, endow: endowEle, provoke: provLv,
    overthrust: bv('BS_OVERTHRUST') ? {self: can('BS_OVERTHRUST'), power: can('BS_OVERTHRUST') ? 5 * bv('BS_OVERTHRUST') : 5} : null,   // 11599: val3 = self ? 5 * lv : 5
    adrenaline: bv('BS_ADRENALINE') ? {self: can('BS_ADRENALINE')} : null, songs: song,
    volcano: bv('SA_VOLCANO') ? {lv: bv('SA_VOLCANO'), atk: volcanoAtk, enchant: FIELD_ENCHANT[(bv('SA_VOLCANO') - 1) % 5], armorOk: fieldOk('Fire')} : null,
    gale: bv('SA_VIOLENTGALE') ? {lv: bv('SA_VIOLENTGALE'), flee: galeFlee, enchant: FIELD_ENCHANT[(bv('SA_VIOLENTGALE') - 1) % 5], armorOk: fieldOk('Wind')} : null,
    deluge: bv('SA_DELUGE') ? {lv: bv('SA_DELUGE'), hpRate: delugeHp, enchant: FIELD_ENCHANT[(bv('SA_DELUGE') - 1) % 5], armorOk: fieldOk('Water')} : null,
    drum: drumAtk ? {atk: drumAtk, def: drumDef} : null, nibelungen: nibAtk ? {atk: nibAtk, hands: ['right', 'left'].filter(h => wInfo[h] && wInfo[h].wlv === 4)} : null,
    defender: defender ? {longReduction: 5 + 15 * defLv, aspdPenalty: (250 - 50 * defLv) / 10} : null};   // 11317: val2 = 5 + 15 * lv (% less BF_LONG|BF_WEAPON damage taken, battle.cpp:1832-1839)
  out.riding = riding; out.falcon = falcon; out.armorEle = armorEle; out.regen = regen;
  out.items = items; out.notes = notes;
  if (out.twoHanded && lItem) notes.push('A two-handed weapon cannot be combined with a left-hand item.');
  return out;
}

/* ---------- validate ---------- */
function validate(build, D) {
  const CD = D.CALCDATA, P = [], job = build.job, J = CD.jobs[job];
  if (!J) return ['Unknown job ' + job];
  const baseLv = build.baseLv | 0, jobLv = build.jobLv | 0;
  if (baseLv < 1 || baseLv > 99) P.push(`Base level ${baseLv} is outside 1..99.`);
  if (jobLv < 1 || jobLv > J.maxJobLv) P.push(`Job level ${jobLv} is outside 1..${J.maxJobLv} for ${jobDisplay(job)}.`);
  const st = build.stats || {};
  for (const k of ST) { const v = st[k]; if (!(v >= 1 && v <= 99) || v !== T(v)) P.push(`${k.toUpperCase()} ${v} is outside 1..99.`); }
  const used = statPointsUsed(st), total = CD.statPoints[clamp(baseLv, 1, 99) - 1];
  if (used > total) P.push(`Stat points: ${used} used, only ${total} available at base level ${baseLv}.`);
  const eq = build.equip || {}, sex = job === 'Bard' ? 'Male' : job === 'Dancer' ? 'Female' : (build.settings && build.settings.sex) || null;
  const itemOf = e => (e && D.items[e.key]) || null;
  const nm = (s, i) => `${s} (${i.name || i.key})`;
  const rItem = itemOf(eq.weapon), lItem = itemOf(eq.left);
  for (const slot of SLOTS) {
    const e = eq[slot]; if (!e) continue;
    const item = D.items[e.key];
    if (!item) { P.push(`${slot}: unknown item ${e.key}.`); continue; }
    const tag = nm(slot, item);
    // position
    const loc = item.loc;
    let fits = false;
    if (slot === 'weapon') fits = item.type === 'Weapon' && (loc === 'Weapon' || loc === 'Two-handed Weapon');
    else if (slot === 'left') fits = (item.type === 'Armor' && loc === 'Shield') || (item.type === 'Weapon' && loc === 'Weapon');
    else if (slot === 'headTop' || slot === 'headMid' || slot === 'headLow') {
      const cov = headCover(item);
      fits = !!cov && item.type === 'Armor';
      if (fits && headSlotOf(cov) !== slot) { P.push(`${tag}: ${loc} belongs in ${headSlotOf(cov)}.`); fits = true; }
      if (fits && headSlotOf(cov) === slot) for (const c of cov.slice(1)) { const o = c === 'mid' ? 'headMid' : 'headLow'; if (eq[o]) P.push(`${tag}: covers the ${o} position, which is occupied.`); }
    } else if (slot === 'armor') fits = item.type === 'Armor' && loc === 'Armor';
    else if (slot === 'garment') fits = item.type === 'Armor' && loc === 'Garment';
    else if (slot === 'shoes') fits = item.type === 'Armor' && loc === 'Footgear';
    else if (slot === 'acc1' || slot === 'acc2') fits = item.type === 'Armor' && loc === 'Accessory';
    else if (slot === 'ammo') fits = item.type === 'Ammunition';
    if (!fits) { P.push(`${tag}: cannot be worn in the ${slot} position.`); continue; }
    // job / sex / level
    if (item.jobs && item.jobs.length && item.jobs.indexOf(jobDisplay(job)) < 0) P.push(`${tag}: ${jobDisplay(job)} cannot equip it.`);
    else if (item.type === 'Weapon' && J.aspd[weaponType(item)] == null && weaponType(item) !== 'Fist') P.push(`${tag}: ${jobDisplay(job)} has no ASPD entry for ${weaponType(item)}.`);
    if (sex && item.sex && item.sex !== 'Both' && item.sex !== sex) P.push(`${tag}: only for ${item.sex} characters.`);
    if (item.eqlv > baseLv) P.push(`${tag}: needs base level ${item.eqlv}.`);
    // refine / forge / cards
    const refine = e.refine | 0;
    if (refine < 0 || refine > 10) P.push(`${tag}: refine +${refine} is outside 0..10.`);
    else if (refine > 0 && !item.refine) P.push(`${tag}: cannot be refined.`);
    const forged = !!e.forgedEle || (e.starCrumbs | 0) > 0;
    if (forged && !((slot === 'weapon' || slot === 'left') && item.type === 'Weapon')) P.push(`${tag}: only weapons can be forged.`);
    if (e.forgedEle && ELES.indexOf(e.forgedEle) < 0) P.push(`${tag}: unknown forged element ${e.forgedEle}.`);
    if ((e.starCrumbs | 0) < 0 || (e.starCrumbs | 0) > 3) P.push(`${tag}: star crumbs must be 0..3.`);
    const cards = (e.cards || []).filter(Boolean);
    if (cards.length > (item.slots | 0)) P.push(`${tag}: ${cards.length} cards but only ${item.slots | 0} slots.`);
    if (forged && cards.length) P.push(`${tag}: a forged weapon cannot hold cards.`);
    for (const ck of cards) {
      const c = D.items[ck];
      if (!c || c.type !== 'Card') { P.push(`${tag}: ${ck} is not a card.`); continue; }
      let want = CARDSLOT[slot];
      if (slot === 'left' && item.type === 'Weapon') want = 'Weapon';
      if (c.sub !== 'Card' && c.sub !== want) P.push(`${tag}: ${c.name} is a ${c.sub} card, this is a ${want} slot.`);
    }
  }
  // hand rules
  if (rItem && rItem.loc === 'Two-handed Weapon' && eq.left) P.push('A two-handed weapon cannot be combined with a left-hand item.');
  if (lItem && lItem.type === 'Weapon') {
    if (job !== 'Assassin') P.push(`Only Assassins can dual wield; ${jobDisplay(job)} needs a shield in the left hand.`);
    else if (!rItem) P.push('A left-hand weapon needs a right-hand weapon.');
    else if (['Dagger', '1hSword', '1hAxe'].indexOf(weaponType(lItem)) < 0) P.push(`A ${weaponType(lItem)} cannot be used in the left hand.`);
  }
  return P;
}

return {compute, validate, BUFFS, weaponType, SONG_CASTER, statPointsUsed, ST, SLOTS, manualBonus, MANUAL_NUM: MANUAL_NUM.map(x => x[0]), MANUAL_MAP: MANUAL_MAP.map(x => x[0])};
});
